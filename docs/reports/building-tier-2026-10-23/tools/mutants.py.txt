#!/usr/bin/env python3
"""mutants.py: each mutant must make `structcount.py selftest` fail for its named case. Anchors are asserted present and
unique before applying (CLAUDE.md: a mutant that silently fails to apply reads as killed). Run from the tools directory."""
import subprocess, tempfile, os
src = open('structcount.py.txt').read()
muts = [('house joined to a buried stair', "        rooms = rooms_of(w, comp, built)\n", "        rooms = rooms_of(w, comp, built) if bur <= 0.25 else []\n"),
        ('unlit lamp only', "    return bool(LIGHT_IF_LIT.match(n)) and str(w.props(x, y, z).get('lit', 'false')) == 'true'", "    return bool(LIGHT_IF_LIT.match(n))"),
        ('light below the floor only', "            if lit_at(w, x, y, z) or lit_at(w, x, y + 1, z): lit = True; break",
         "            if lit_at(w, x, y, z) or lit_at(w, x, y + 1, z) or lit_at(w, x, y - 2, z): lit = True; break")]
for case, old, new in muts:
    assert src.count(old) == 1, 'ANCHOR MISSING OR NOT UNIQUE: ' + case
    f = tempfile.NamedTemporaryFile('w', suffix='.py', delete=False); f.write(src.replace(old, new, 1)); f.close()
    out = subprocess.run(['python3', f.name, 'selftest'], capture_output=True, text=True).stdout
    os.unlink(f.name)
    killed = any(l.startswith(case) and l.rstrip().endswith('FAIL') for l in out.splitlines())
    print('%-32s %s' % (case, 'KILLED' if killed else 'SURVIVED'))
