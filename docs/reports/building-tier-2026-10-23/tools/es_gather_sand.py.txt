import sys, re, collections, json
sys.path.insert(0, '/home/mike/estel/scripts')
from lib.estelemetry import EsEvents, ZeroLooksWrong
UNTIL = '2026-10-23T14:00:00Z'
ev = EsEvents.load('skill', since_minutes=24 * 60, until=UNTIL)
# gather by target (args.target or detail)
rows = ev.rows('gather', fields=['skill.status', 'skill.detail', 'skill.args', 'skill.fail_class', 'bot.name', 'exp.pool'], max_rows=60000)
tgt = collections.Counter(); fails = collections.Counter(); samp = {}
bots = collections.defaultdict(set)
n = 0
for r in rows:
    n += 1
    sk = r.get('skill') or {}
    a = sk.get('args') or {}
    t = (a.get('target') or a.get('block') or a.get('name') or a.get('item')) if isinstance(a, dict) else None
    d = sk.get('detail') or ''
    if not t:
        m = re.search(r'\b(sand|red_sand|clay|gravel)\b', d)
        t = m.group(1) if m else '?'
    tgt[(t, sk.get('status'))] += 1
    if t in ('sand', 'red_sand'):
        key = re.sub(r'\d+', 'N', d)[:110]
        fails[(sk.get('status'), sk.get('fail_class'), key)] += 1
        bots[t].add((r.get('bot') or {}).get('name'))
print('gather rows walked', n)
for k, v in tgt.most_common(30):
    print('TGT', k, v)
print('sand bots', {k: len(v) for k, v in bots.items()})
for k, v in fails.most_common(25):
    print('SAND', v, k)
