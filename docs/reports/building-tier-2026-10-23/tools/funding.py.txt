#!/usr/bin/env python3
"""funding.py <census-species.jsonl> -- which towns can fund the shared building and then houses, one instance at a time,
from SURPLUS ABOVE THE RESERVE (OWNER B3). Chests only (a bag is not the town's). r3 (Codex r2 P1, Claude r2 P3):

  * the bill is built from each blueprint's CELLS BY ROLE (§4.2), with the role's fallback chosen per instance:
      trim  -> cut_copper (9 ingots each, 4 at a time from 4 copper blocks) or its fallback (workshop roof edge: planks;
               house corners: stone bricks);
      window-> glass_pane (16 per 6 sand) or its fallback (planks) when sand does not allocate;
      door  -> copper_door (6 ingots -> 3) or an open doorway;
  * stone bricks come from COBBLESTONE only (cobbled deepslate smelts to deepslate), in sessions of 12 cobble on exactly 8
    planks; bricks are crafted 4 at a time; a glass session is 6 sand on 4 planks;
  * wall planks are of ONE species, the town's most-stocked; fuel planks may be any; banked planks are used before logs;
  * stations (1 table: 4 planks; 2 furnaces: 8 cobblestone each), torches (from stock above the torch reserve) and the
    earthwork fill allowance (40 cobblestone per pad) are in the bill;
  * ADMISSION is TP 3.2's: the instance's WHOLE bill must allocate within 80% of what is spendable for EVERY input;
  * leftovers (bricks over the need, panes, cut copper, doors) stay in a parts pool and are used by the next instance;
    banked parts (none today) are used first;
  * isolated-a, c, d are OUT (TP D2 / town plan O8: per-bot town state). isolated-b is a shared town since 10-21."""
import json, math, sys

OUT = {'isolated-a', 'isolated-c', 'isolated-d'}
RESERVE = {'cobblestone': 256, 'logs': 512, 'planks': 64, 'copper_ingot': 256, 'sand': 16, 'torch': 32}

WORKSHOP = {'bricks': 63 + 26, 'wall_planks': 34, 'roof_planks': 15, 'windows': 4, 'trim': 20, 'trim_fallback': 'planks',
            'torches': 2, 'door': 0, 'tables': 1, 'furnaces': 2, 'fill': 40}
HOUSE = {'bricks': 49 + 23, 'wall_planks': 27, 'roof_planks': 21, 'windows': 3, 'trim': 4, 'trim_fallback': 'bricks',
         'torches': 1, 'door': 1, 'tables': 0, 'furnaces': 0, 'fill': 40}

def bill(bp, sp, pool):
    """The raw draw of ONE instance, after the parts pool, with fallbacks chosen against spendable stock. Returns
    (draw, new_pool, choices). draw keys: cobblestone, wall_logs, any_logs, planks, sand, copper_ingot, torch."""
    pool = dict(pool); ch = {}
    def tryparts(name, n):
        use = min(pool.get(name, 0), n); pool[name] = pool.get(name, 0) - use; return n - use
    cu_need = 9 * 4 * math.ceil(bp['trim'] / 4) if bp['trim'] else 0
    door_cu = 6 if bp['door'] and pool.get('copper_door', 0) < bp['door'] else 0
    ch['trim'] = 'copper' if (cu_need + door_cu) <= 0.8 * sp['copper_ingot'] else bp['trim_fallback']
    ch['door'] = 'copper_door' if bp['door'] and ch['trim'] == 'copper' else ('doorway' if bp['door'] else '-')
    pane_need = tryparts('glass_pane', bp['windows'])
    sand = 6 * math.ceil(pane_need / 16) if pane_need else 0
    ch['windows'] = 'glass_pane' if sand <= 0.8 * sp['sand'] else 'planks'
    bricks = bp['bricks'] + (bp['trim'] if ch['trim'] == 'bricks' else 0)
    bricks_new = tryparts('stone_bricks', bricks)
    b4 = 4 * math.ceil(bricks_new / 4)
    sessions = math.ceil(b4 / 12)
    pool['stone_bricks'] = pool.get('stone_bricks', 0) + (12 * sessions - bricks_new)     # leftover bricks
    wall_planks = bp['wall_planks'] + (bp['trim'] if ch['trim'] == 'planks' else 0) + (bp['windows'] if ch['windows'] == 'planks' else 0)
    roof_planks = bp['roof_planks']
    fuel_planks = 8 * sessions + (4 if (ch['windows'] == 'glass_pane' and sand) else 0)
    station_planks = 4 * bp['tables']
    draw = {'cobblestone': 12 * sessions + 8 * bp['furnaces'] + bp['fill'],
            'wall_planks': wall_planks + roof_planks + station_planks, 'fuel_planks': fuel_planks,
            'sand': sand if ch['windows'] == 'glass_pane' else 0,
            'copper_ingot': (cu_need + door_cu) if ch['trim'] == 'copper' else 0, 'torch': bp['torches']}
    if ch['windows'] == 'glass_pane' and sand:
        pool['glass_pane'] = pool.get('glass_pane', 0) + 16 * math.ceil(pane_need / 16) - pane_need
    if ch['trim'] == 'copper':
        pool['cut_copper'] = pool.get('cut_copper', 0) + 4 * math.ceil(bp['trim'] / 4) - bp['trim']
        if door_cu: pool['copper_door'] = pool.get('copper_door', 0) + 3 - bp['door']
        elif bp['door']: pool['copper_door'] -= bp['door']
    return draw, pool, ch

def spendable(t):
    logs_total = sum(t['logs'].values()); planks_total = sum(t['planks'].values())
    main = max(t['logs'], key=t['logs'].get) if t['logs'] else 'oak'
    return {'cobblestone': t['cobblestone'] - RESERVE['cobblestone'], 'logs': logs_total - RESERVE['logs'],
            'main_logs': t['logs'].get(main, 0), 'main': main, 'planks': planks_total - RESERVE['planks'],
            'sand': t['sand'] - RESERVE['sand'], 'copper_ingot': t['copper_ingot'] - RESERVE['copper_ingot'],
            'torch': t['torch'] - RESERVE['torch']}

def admits(sp, d):
    """TP 3.2: every input of the whole bill within 80% of spendable. Planks: banked planks first, then logs (4 each); the wall
    planks must come from the main species' logs or banked planks."""
    planks_need = d['wall_planks'] + d['fuel_planks']
    from_banked = min(max(sp['planks'], 0), planks_need)
    logs_need = math.ceil((planks_need - from_banked) / 4)
    wall_logs_need = math.ceil(max(0, d['wall_planks'] - from_banked) / 4)
    ok = {'cobblestone': d['cobblestone'] <= 0.8 * sp['cobblestone'],
          'logs': logs_need <= 0.8 * sp['logs'] and wall_logs_need <= 0.8 * sp['main_logs'],
          'sand': d['sand'] <= 0.8 * sp['sand'] if d['sand'] else True,
          'copper_ingot': d['copper_ingot'] <= 0.8 * sp['copper_ingot'] if d['copper_ingot'] else True,
          'torch': d['torch'] <= 0.8 * sp['torch']}
    return all(ok.values()), ok, logs_need

def spend(sp, d, logs_need):
    sp = dict(sp)
    sp['cobblestone'] -= d['cobblestone']; sp['logs'] -= logs_need; sp['main_logs'] -= min(sp['main_logs'], logs_need)
    sp['planks'] -= min(max(sp['planks'], 0), d['wall_planks'] + d['fuel_planks']); sp['sand'] -= d['sand']
    sp['copper_ingot'] -= d['copper_ingot']; sp['torch'] -= d['torch']
    return sp

towns = [json.loads(l) for l in open(sys.argv[1])]
n = wf = wc = 0; houses = []
for t in towns:
    if t['world'] in OUT:
        print('%-11s OUT (isolated, per-bot town state: D2)' % t['world']); continue
    n += 1
    sp = spendable(t); pool = dict(t.get('parts') or {})
    d, pool, ch = bill(WORKSHOP, sp, pool)
    ok, why, logs_need = admits(sp, d)
    nh = 0
    if ok:
        wf += 1; wc += ch['trim'] == 'copper'
        sp = spend(sp, d, logs_need)
        while nh < 5:
            hd, hpool, hch = bill(HOUSE, sp, pool)
            hok, hwhy, hlogs = admits(sp, hd)
            if not hok: break
            sp = spend(sp, hd, hlogs); pool = hpool; nh += 1
    houses.append(nh)
    print('%-11s workshop %-4s trim=%-7s windows=%-10s | bill %s | short %s | houses after it, one at a time: %d' % (
        t['world'], 'YES' if ok else 'NO', ch['trim'], ch['windows'], d, [k for k, v in why.items() if not v] or '-', nh))
print('of %d eligible towns: workshop admitted in %d (%d with copper trim); houses admissible in sequence from today\'s stock, '
      'no refill: %d of %d (per town %s). Sand is 0 everywhere today, so every window is the plank fallback until matset-01.'
      % (n, wf, wc, sum(houses), 5 * n, houses))
