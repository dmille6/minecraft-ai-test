# comms design 2026-10-21 r5 (deaths_v5): merge-before-prune on STORED counts, two write-order extremes A and C.
# (r4 header kept below for history)
# comms design 2026-10-21 r4: death-site replay with EXPLICIT query bounds and a sensitivity bound for write ordering.
# Production (worldfacts.mjs reportDeath + #prune, deathsites.mjs effectiveCount): a death merges into a same-class site
# within 3 blocks / 3 y of its kept centre (+4 weight, deaths+1), else a new site (weight 4, JS Math.round centre).
# The file is pruned (decay applied, weight < 1 dropped) on EVERY write; world facts is written every few seconds, so a
# decayed site is normally gone before the next death -- variant A (decay, then merge only into live sites).
# Variant B bounds the other case (no write between the decay and the death): a merge may revive a site whose decayed
# weight is 0 if it had not been pruned yet -- here: any site seen in the last 72 h. The 120-site cap (shared with
# hazards) is NOT replayed: eviction could change which centres survive, so both variants are APPROXIMATE, not bounds
# on production. Read-only.
import sys, os, json, math, collections, glob, datetime as dt
sys.path.insert(0, os.path.expanduser('~/estel/scripts'))
from lib.estelemetry import EsEvents
POOLS = [f'{a}-{b}' for a in ('board', 'hive', 'placebo') for b in 'abcd']
H = 3600; DECAY = 12 * H
UNTIL = dt.datetime(2026, 10, 21, 20, 0, tzinfo=dt.timezone.utc)
SINCE = UNTIL - dt.timedelta(days=9); JUDGE_FROM = SINCE + dt.timedelta(days=2)
ev = EsEvents.load(stream='skill', since=SINCE.isoformat(), until=UNTIL.isoformat(), pools=POOLS)
print('query', SINCE.isoformat(), '..', UNTIL.isoformat(), '; judged', JUDGE_FROM.isoformat(), '..', UNTIL.isoformat(), '(7.0 days)')
print('caveats:', *ev.caveats, sep='\n  ')
rows = list(ev.rows(kind='_death', fields=['@timestamp', 'bot.name', 'bot.pos', 'exp.pool', 'skill.fail_class']))
def ts(s): return dt.datetime.fromisoformat(s.replace('Z', '+00:00')).timestamp()
D = sorted(({'t': ts(r['@timestamp']), 'bot': r['bot']['name'], 'pool': r['exp']['pool'],
             'p': (r['bot']['pos']['x'], r['bot']['pos']['y'], r['bot']['pos']['z']),
             'fc': (r.get('skill') or {}).get('fail_class') or 'unknown'} for r in rows if (r.get('bot') or {}).get('pos')), key=lambda d: d['t'])
print('_death rows', len(rows), 'with position', len(D))
def jsround(v): return math.floor(v + 0.5)
def weight(s, now):
    ref = max(s['last'], s.get('dec', 0)); per = int((now - ref) // DECAY) if now > ref else 0
    return s['count'] // (2 ** per)
def prune(S, now):
    # worldfacts.mjs #prune: decay every whole elapsed period since max(last, decayedAt), drop weight < 1
    keep = []
    for s in S:
        ref = max(s['last'], s.get('dec', 0)); per = int((now - ref) // DECAY) if now > ref else 0
        if per > 0: s['count'] //= 2 ** per; s['dec'] = ref + per * DECAY
        if s['count'] >= 1: keep.append(s)
    S[:] = keep
def run(variant):
    # A: the file is written (pruned) just before every death -- world facts is rewritten every few seconds.
    # C: NO write between deaths in a town -- the file is pruned only by the death writes themselves.
    # Both: reportDeath reads the STORED sites, merges (+4 to the stored count, last = now), then #prune runs.
    # Production lies between these two write orders.
    sites = collections.defaultdict(list); res = collections.Counter(); conf = []; cls = collections.Counter()
    tj = JUDGE_FROM.timestamp()
    for d in D:
        S = sites[d['pool']]
        if variant == 'A': prune(S, d['t'])
        live = [s for s in S if weight(s, d['t']) >= 1]          # deathSites() applies the decay on read
        cover = [s for s in live if math.hypot(s['x'] - d['p'][0], s['z'] - d['p'][2]) <= 6 and abs(s['y'] - d['p'][1]) <= 6]
        if d['t'] >= tj:
            res['judged'] += 1; cls[d['fc']] += 1
            if not cover: res['no_live_site'] += 1
            elif any(set(s['by']) - {d['bot']} for s in cover): res['site_with_another_bots_death'] += 1
            else: res['own_only'] += 1
            if d['fc'] == 'fire':
                res['fire_judged'] += 1
                if [s for s in cover if s['fc'] == 'fire']: res['fire_in_live_fire_site'] += 1
                if [s for s in cover if s['fc'] == 'fire' and s['deaths'] >= 2 and len(set(s['by'])) >= 2]:
                    res['fire_in_confirmed_site'] += 1; conf.append(f"{dt.datetime.fromtimestamp(d['t'], dt.timezone.utc):%m-%d %H:%M} {d['bot']}")
        m = next((s for s in S if s['fc'] == d['fc'] and math.hypot(s['x'] - d['p'][0], s['z'] - d['p'][2]) <= 3 and abs(s['y'] - d['p'][1]) <= 3), None)
        if m:
            m['count'] += 4; m['deaths'] += 1; m['last'] = d['t']; m['by'].append(d['bot'])
        else:
            S.append({'fc': d['fc'], 'x': jsround(d['p'][0]), 'y': jsround(d['p'][1]), 'z': jsround(d['p'][2]),
                      'count': 4, 'deaths': 1, 'last': d['t'], 'by': [d['bot']]})
        prune(S, d['t'])
    end = UNTIL.timestamp()
    livefire = {p: sum(1 for s in S if s['fc'] == 'fire' and weight(s, end) >= 1) for p, S in sites.items()}
    return res, conf, livefire, cls
for v in ('A', 'C'):
    res, conf, lf, cls = run(v)
    print(f'variant {v}:', dict(res))
    print('   confirmed-site fire deaths:', conf)
    print('   judged deaths by class:', dict(cls))
    print('   live fire sites at', UNTIL.isoformat(), {p: n for p, n in lf.items() if n})
