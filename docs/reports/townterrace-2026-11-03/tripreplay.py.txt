# townterrace-01: replay the terrace stage's own rows (generated from its offline record) through main's shopread.py judge
import json, sys, datetime as dt, importlib.util
spec = importlib.util.spec_from_file_location('shopread', sys.argv[1]); S = importlib.util.module_from_spec(spec); spec.loader.exec_module(S)
d = json.load(open(sys.argv[2]))
rec, rec0 = d['record'], d['stage0']
t0 = dt.datetime(2026, 11, 3, 6, 0, tzinfo=dt.timezone.utc)
R = lambda name, detail, s=0: {'t': t0 + dt.timedelta(seconds=s), 'per': 'post', 'arm': 'canary', 'name': name, 'detail': detail, 'status': 'success', 'bot': 'showcase-Alpha', 'item': None, 'feet': None}
c3 = lambda a: '%d,%d,%d' % tuple(a)
rows = [R('_build_mode', 'phase=mode v=1 shop=on'), R('_build', 'project=tidy op=sited cell=0,70,0 gen=2', s=1)]
need = len([x for x in rec.get('replant') or [] if x.get('state') != 'skipped'])
s = 2
for r in rec.get('replant') or []: rows.append(R('_build', 'project=tidy op=replant cell=%s block=%s' % (c3(r['cell']), r['sapling']), s=s)); s += 1
for c in rec['cut']: rows.append(R('_build', 'project=tidy op=clear cell=%s block=%s kind=tidy' % (c3(c['at']), c['name']), s=s)); s += 1
for c in rec['clear']:
    extra = ' saplings=%d/%d' % (need, need) if c.get('kind') == 'tree' else ''
    rows.append(R('_build', 'project=tidy op=clear cell=%s block=%s kind=%s%s' % (c3(c['at']), c.get('name'), c.get('kind'), extra), s=s)); s += 1
for c in rec['cells']:
    use = c['role'] in ('till', 'meadow')
    rows.append(R('_build', 'project=tidy op=place seg=%s cell=%s block=%s%s' % (c['seg'], c3(c['at']), 'farmland' if c['role'] == 'till' else ('bone_meal' if c['role'] == 'meadow' else 'dirt'), ' kind=use' if use else ''), s=s)); s += 1
# a foreign dig in an open fill cell (dirt on a cell the visit re-places)
fc = next(c for c in rec['cells'] if c['role'] in ('earth', 'core'))
rows.append(R('_build', 'project=tidy op=clear cell=%s block=dirt kind=foreign' % c3(fc['at']), s=s)); s += 1
chain = [x for x in (rec0, rec) if x]
proj = {S.pool_of('showcase-Alpha'): {'record': rec, 'records_files': len(chain), 'leases': {}, 'zones_ok': True, 'markers': {}, 'projects': {'tidy': {'record': rec, 'records_files': len(chain), 'leases': {}, 'chain': chain}}}}
zj = json.load(open(sys.argv[3])) if len(sys.argv) > 3 else []
zl = zj.get('zones', zj) if isinstance(zj, dict) else zj
ZB = {S.pool_of('showcase-Alpha'): [z['box'] for z in zl if isinstance(z, dict) and z.get('box')]}
print('protected zone boxes', len(ZB[S.pool_of('showcase-Alpha')]))
J = lambda rr: S.combine(*S.judge(rr, proj, ZB, cut=None, gens=()))
res = J(rows)
keys = ('off_record', 'clear_outside', 'zone', 'lease_overlap', 'felled_unplanted', 'bag_loss', 'oracle_missing', 'zone_unknown', 'g5_unjudged')
print('rows', len(rows), 'placements', res.get('placements'), 'clears', res.get('clears'), {k: res[k] for k in keys})
assert all(res[k] == 0 for k in keys), res['_lists']
# positive controls: an off-list cut, an off-record place, a tree felled before its sapling
x, y, z = rec['cut'][0]['at']
bad = J(rows + [R('_build', 'project=tidy op=clear cell=%d,%d,%d block=stone kind=tidy' % (x, y + 30, z), s=s)])
assert bad['clear_outside'] == 1, bad['clear_outside']
bad2 = J(rows + [R('_build', 'project=tidy op=place seg=T-PP cell=%d,%d,%d block=poppy' % (x, y + 30, z), s=s)])
assert bad2['off_record'] == 1, bad2['off_record']
tl = next((c for c in rec['clear'] if c.get('kind') == 'tree'), None)
if tl and need:
    bad3 = J(rows + [R('_build', 'project=tidy op=clear cell=%s block=%s kind=tree saplings=0/%d' % (c3(tl['at']), tl['name'], need), s=s)])
    assert bad3['felled_unplanted'] == 1, bad3['felled_unplanted']
print('positive controls: off-list cut H2 1, off-record place H1 1, felled before sapling H5 1 -- OK')
