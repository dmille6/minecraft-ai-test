import sys, os, datetime as dt, collections as C, json, re
sys.path.insert(0, '.')
from lib import telemetry as TM, effectmetrics as EM
UTC=dt.timezone.utc
lo=dt.datetime.fromisoformat(sys.argv[1]).replace(tzinfo=UTC); hi=dt.datetime.fromisoformat(sys.argv[2]).replace(tzinfo=UTC)
root='/var/log/mcai'
bots=sorted(d for d in os.listdir(root) if not d.startswith('_'))
per=C.defaultdict(C.Counter); keys=C.defaultdict(C.Counter); det=C.defaultdict(C.Counter); runs=C.Counter(); prev=C.Counter()
ysamp=C.defaultdict(list); ms=C.defaultdict(C.Counter); calls=C.Counter()
for b in bots:
    rows=sorted(TM.stream_rows(root,b,lo,hi,'llm'), key=lambda x:x[0])
    run=0; last=None
    for t,r in rows:
        if not EM.is_model_call(r): continue
        calls[b]+=1
        e=(r.get('llm') or {}).get('error'); tc=r.get('tool_calls') or [{}]; tc=tc[0] if tc else {}
        sk=tc.get('skill'); a=tc.get('args') or {}
        if e and not str(e).startswith('trunc'):
            per[e][b]+=1
            k=f"{sk} {a.get('block') or a.get('item') or a.get('target') or ''}".strip()
            keys[e][k]+=1
            d=re.sub(r'\d+','N',str((r.get('outcome') or {}).get('detail') or ''))[:110]
            det[e][d]+=1
            ms[e][(r.get('messages') or [{}])[0].get('milestone')]+=1
            ysamp[e].append(((r.get('bot') or {}).get('pos') or {}).get('y'))
            prev[(last or 'start', e)]+=1
            last=e; run+=1
        else:
            if run: runs[min(run,10)]+=1
            run=0; last='ok'
tot=sum(calls.values()); print('model calls',tot,'bots',len(calls))
for e,c in sorted(per.items(), key=lambda kv:-sum(kv[1].values()))[:8]:
    n=sum(c.values()); top=c.most_common(10)
    print(f'\n== {e}: {n} ({100*n/tot:.1f}% of calls), {len(c)} bots; top10 bots {sum(v for _,v in top)} ({100*sum(v for _,v in top)/n:.0f}%): {top[:5]}')
    print('  keys', keys[e].most_common(6))
    print('  milestones', ms[e].most_common(5))
    ys=[y for y in ysamp[e] if y is not None]; ys.sort(); print('  y quartiles', [round(ys[int(q*(len(ys)-1))]) for q in (0,.25,.5,.75,1)] if ys else None)
    for d,v in det[e].most_common(3): print('  ',v,d)
print('\nveto run lengths (consecutive vetoed calls)', sorted(runs.items()))
print('prev->veto', prev.most_common(15))
