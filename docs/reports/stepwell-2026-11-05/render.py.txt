# stepwell-01's pictures (stdlib only): render.py <grid.json> <out-dir>
# grid.json is bots/scripts/stepwell-offline.mjs --grid over a read-only copy of the showcase's region files. Draws:
#   top.png      the site from above: the natural ground round it, the tank's columns by depth, posts, lights, flights, the basin
#   section.png  a side cross-section through the centre, y -64 .. g + 12: the real blocks, the tank cut into them (tiers 1-2 built
#                by piece 1, tier 3 dashed), the phase-1b/2 descent below the court (schematic) and the sealed lava station
#   iso.png      an isometric view of the finished tank (tiers 1-2) and the ground round it
import json, sys, zlib, struct, math, os

g = json.load(open(sys.argv[1])); out = sys.argv[2]
os.makedirs(out, exist_ok=True)
site = g['site']; G = site['g']; W = site['W']


def png(path, img):
    h = len(img); w = len(img[0])
    raw = b''.join(b'\x00' + bytes(v for p in row for v in p) for row in img)
    def chunk(t, d): return struct.pack('>I', len(d)) + t + d + struct.pack('>I', zlib.crc32(t + d) & 0xffffffff)
    open(path, 'wb').write(b'\x89PNG\r\n\x1a\n' + chunk(b'IHDR', struct.pack('>IIBBBBB', w, h, 8, 2, 0, 0, 0)) + chunk(b'IDAT', zlib.compress(raw, 9)) + chunk(b'IEND', b''))
    print(path, w, h)


def canvas(w, h, c=(250, 250, 248)): return [[c] * w for _ in range(h)]


def rect(img, x0, y0, x1, y1, c):
    for y in range(max(0, y0), min(len(img), y1)):
        row = img[y]
        for x in range(max(0, x0), min(len(row), x1)): row[x] = c


def shade(c, f): return tuple(max(0, min(255, int(v * f))) for v in c)

def text(img, x, y, s, c=(20, 20, 20), k=2):
    for ch in s.upper():
        bits = FONT.get(ch, FONT[' '])
        for i, bit in enumerate(bits):
            if bit == '1': rect(img, x + (i % 3) * k, y + (i // 3) * k, x + (i % 3) * k + k, y + (i // 3) * k + k, c)
        x += 4 * k


KIND = {'walk': (196, 196, 190), 'rail': (110, 110, 110), 'court': (205, 200, 190), 'basin': (90, 140, 210), 'gap': (196, 196, 190),
        'head': (150, 120, 90), 'tread': (176, 140, 100), 'frail': (130, 105, 80), 'parapet': (110, 110, 110), 'gate': (196, 196, 190)}


def natcol(n, y):
    if n == '?' or y is None: return (40, 40, 40)
    if 'water' in n: return (70, 110, 200)
    if n.endswith('leaves'): return (40, 110, 50)
    if n.endswith('log'): return (110, 80, 40)
    if 'grass' in n or n in ('podzol', 'moss_block'): base = (110, 160, 80)
    elif 'dirt' in n: base = (140, 110, 70)
    elif 'snow' in n: base = (235, 240, 245)
    else: base = (150, 150, 150)
    return shade(base, 0.75 + 0.03 * (y - G))

        n = cmap.get((c['x'] + dx, c['z'] + dz))

def blockcol(n):
    if n == '?': return (40, 40, 40)
    if n in ('air', 'cave_air', 'void_air'): return None
    if 'lava' in n: return (240, 110, 20)
    if 'water' in n or 'kelp' in n or 'seagrass' in n: return (70, 110, 200)
    if 'diamond' in n: return (80, 230, 230)
    if 'iron_ore' in n: return (210, 160, 120)
    if 'coal' in n: return (30, 30, 30)
    if 'copper' in n: return (200, 120, 70)
    if n.endswith('_ore'): return (230, 200, 60)
    if n == 'obsidian': return (50, 20, 70)
    if n.endswith('leaves'): return (40, 110, 50)
    if n.endswith('log'): return (110, 80, 40)
    if 'deepslate' in n or n == 'tuff': return (80, 82, 90)
    if 'dirt' in n or 'grass' in n or n in ('podzol', 'gravel', 'sand', 'clay'): return (140, 110, 70)
    if n == 'bedrock': return (20, 20, 20)
    return (140, 140, 140)

# ---------------------------------------------------------------- isometric
# voxels: the tank as built (tiers 1-2): floors / trims (stone), posts (wall), lights (torch), the ground round it (natural top)
vox = {}
for c in g['cols']:
    u, v, s = c['u'], c['v'], c['s']
    base = G - 10
    for y in range(base, s + 1):
        vox[(u, y, v)] = (182, 182, 176) if y == s else (150, 146, 140)
    if c['post']: vox[(u, s + 1, v)] = (96, 96, 96)
    if c['light']: vox[(u, s + 2, v)] = 'torch'
for q in g['nat']:
    # the ring of ground outside the footprint (local coordinates from the site's frame: away = n -> u = x - ox reversed)
    pass
A = 12  # voxel width
B = A // 2
Hh = A * 3 // 4
def iso(u, y, v): return ((u - v) * A, (u + v) * B - y * Hh)
pts = [iso(u, y, v) for (u, y, v) in vox]
mx = min(p[0] for p in pts) - A; my = min(p[1] for p in pts) - 2 * A
Wd = max(p[0] for p in pts) - mx + 2 * A; Ht = max(p[1] for p in pts) - my + 3 * A
img = canvas(Wd, Ht + 30, (235, 240, 246))


def poly(img, ps, c):
    ys = [p[1] for p in ps]
    for yy in range(int(min(ys)), int(max(ys)) + 1):
        xs = []
        for i in range(len(ps)):
            (xa, ya), (xb, yb) = ps[i], ps[(i + 1) % len(ps)]
            if ya == yb: continue
            if min(ya, yb) <= yy < max(ya, yb): xs.append(xa + (yy - ya) * (xb - xa) / (yb - ya))
        xs.sort()
        for j in range(0, len(xs) - 1, 2):
            if 0 <= yy < len(img):
                for xx in range(int(xs[j]), int(xs[j + 1]) + 1):
                    if 0 <= xx < len(img[0]): img[yy][xx] = c


for (u, y, v) in sorted(vox, key=lambda k: (k[0] + k[2], k[1], k[0])):
    c = vox[(u, y, v)]
    X, Y = iso(u, y, v); X -= mx; Y -= my
    if c == 'torch':
        # a torch: a thin stick and a flame, not a block
        rect(img, X - 1, Y + B - 2, X + 2, Y + 2 * B + Hh - 2, (120, 90, 50))
        rect(img, X - 2, Y + B - 6, X + 3, Y + B - 1, (255, 200, 40))
        continue
    topf = [(X, Y), (X + A, Y + B), (X, Y + 2 * B), (X - A, Y + B)]
    if vox.get((u, y + 1, v), 'torch') == 'torch': poly(img, topf, c)
    if vox.get((u, y, v + 1), 'torch') == 'torch': poly(img, [(X - A, Y + B), (X, Y + 2 * B), (X, Y + 2 * B + Hh), (X - A, Y + B + Hh)], shade(c, 0.78))
    if vox.get((u + 1, y, v), 'torch') == 'torch': poly(img, [(X, Y + 2 * B), (X + A, Y + B), (X + A, Y + B + Hh), (X, Y + 2 * B + Hh)], shade(c, 0.62))
text(img, 6, Ht + 6, 'STEPWELL TIERS 1-2 ISOMETRIC  POSTS = COBBLESTONE WALLS  YELLOW = TORCHES')
png(os.path.join(out, 'iso.png'), img)
