#!/usr/bin/env python3
"""townmap.py <worlds.json> <overlays.json> <outdir> -- READ-ONLY town census + overhead "before" picture per world.

Runs on the worlds host (10.0.0.30) under nice. For each world it COPIES the region and entity files that cover home +-R
into a scratch directory (cp; the live files are only read) and reads the copies:
  * the surface (top non-air block per column) of the (2R+1)^2 box, rendered top-down at SCALE px/block (north up,
    east right), hill-shaded;
  * every town block of interest in the box at any height, with its y: containers (with the lid test and the double-chest
    half), stations, light, paths, farmland, saplings, ladders;
  * every entity id in the box (item frames, armour stands, item drops, animals) -- the positive control for "no frames".
Markers are drawn from the WORLD's own blocks, plus the well caps / tree-farm cells from the bots' records (overlays.json,
pulled read-only from 10.0.0.31 /var/lib/mcai). Nothing is written to any world file. Pure python (zlib + struct).
"""
import json, os, re, shutil, struct, sys, zlib, math, collections

R = 48
SCALE = 4

# ------------------------------------------------------------------ NBT / Anvil (same format as scripts/lib/anvil.py)
def read_nbt(b):
    i = 0
    def payload(t):
        nonlocal i
        if t == 1: v = struct.unpack_from('>b', b, i)[0]; i += 1; return v
        if t == 2: v = struct.unpack_from('>h', b, i)[0]; i += 2; return v
        if t == 3: v = struct.unpack_from('>i', b, i)[0]; i += 4; return v
        if t == 4: v = struct.unpack_from('>q', b, i)[0]; i += 8; return v
        if t == 5: v = struct.unpack_from('>f', b, i)[0]; i += 4; return v
        if t == 6: v = struct.unpack_from('>d', b, i)[0]; i += 8; return v
        if t == 7: n = struct.unpack_from('>i', b, i)[0]; i += 4; v = b[i:i + n]; i += n; return v
        if t == 8: n = struct.unpack_from('>H', b, i)[0]; i += 2; v = b[i:i + n].decode('utf-8', 'replace'); i += n; return v
        if t == 9:
            et = b[i]; i += 1; n = struct.unpack_from('>i', b, i)[0]; i += 4
            return [payload(et) for _ in range(n)]
        if t == 10:
            d = {}
            while True:
                tt = b[i]; i += 1
                if tt == 0: return d
                n = struct.unpack_from('>H', b, i)[0]; i += 2; k = b[i:i + n].decode('utf-8', 'replace'); i += n
                d[k] = payload(tt)
        if t == 11: n = struct.unpack_from('>i', b, i)[0]; i += 4; v = struct.unpack_from('>%di' % n, b, i); i += 4 * n; return v
        if t == 12: n = struct.unpack_from('>i', b, i)[0]; i += 4; v = struct.unpack_from('>%dq' % n, b, i); i += 8 * n; return v
        raise ValueError('tag %d' % t)
    t = b[i]; i += 1
    n = struct.unpack_from('>H', b, i)[0]; i += 2 + n
    return payload(t)

_region_cache = {}
def chunk_nbt(rdir, cx, cz):
    rx, rz = cx >> 5, cz >> 5
    p = os.path.join(rdir, 'r.%d.%d.mca' % (rx, rz))
    if p not in _region_cache:
        _region_cache[p] = open(p, 'rb').read() if os.path.exists(p) else b''
    data = _region_cache[p]
    if not data: return None, 'missing_region'
    idx = 4 * ((cx & 31) + (cz & 31) * 32)
    off = int.from_bytes(data[idx:idx + 3], 'big') * 4096
    if off == 0: return None, 'absent'
    ln = int.from_bytes(data[off:off + 4], 'big')
    comp = data[off + 4]
    raw = data[off + 5:off + 4 + ln]
    try:
        if comp == 2: raw = zlib.decompress(raw)
        elif comp == 1: raw = zlib.decompress(raw, 16 + zlib.MAX_WBITS)
        elif comp == 3: pass
        else: return None, 'external'
        return read_nbt(raw), 'ok'
    except Exception as e:
        return None, 'error'

def section_values(bs):
    pal = bs.get('palette') or []
    dat = bs.get('data')
    if dat is None or len(pal) <= 1: return None
    bits = max(4, (len(pal) - 1).bit_length())
    per = 64 // bits; mask = (1 << bits) - 1
    vals = []
    for lo in dat:
        lo &= (1 << 64) - 1
        for k in range(per):
            vals.append((lo >> (k * bits)) & mask)
            if len(vals) >= 4096: break
        if len(vals) >= 4096: break
    return vals

AIR = {'minecraft:air', 'minecraft:cave_air', 'minecraft:void_air', 'minecraft:light', 'minecraft:barrier', 'minecraft:structure_void'}
POI = re.compile(r'^minecraft:(chest|trapped_chest|barrel|furnace|blast_furnace|smoker|crafting_table|composter|torch|wall_torch|lantern|'
                 r'soul_torch|dirt_path|farmland|.*_sapling|ladder|.*_bed|bell|.*_sign|.*_wall_sign|anvil|smithing_table|stonecutter|'
                 r'grindstone|lectern|brewing_stand|cauldron|water_cauldron|hopper|campfire|wheat|carrots|potatoes|beetroots|'
                 r'sugar_cane|bamboo|bee_nest|beehive|oak_log|birch_log|spruce_log|jungle_log|acacia_log|dark_oak_log|cherry_log|mangrove_log)$')
# the lid rule, approximated as bots/src/lidrule.mjs serverLidBlocks: a full cube blocks unless it is glass/leaves/...
NOT_BLOCKING = re.compile(r'(air|glass|leaves|torch|lantern|slab|stairs|fence|_wall$|carpet|flower|sapling|short_grass|tall_grass|^minecraft:grass$|'
                          r'fern|chest|snow$|button|lever|rail|sign|door|pane|bars|ladder|vine|water|lava|_bed$|banner|composter|hopper|'
                          r'bush|tulip|dandelion|poppy|orchid|allium|bluet|daisy|cornflower|lily|rose|peony|lilac|sunflower|mushroom|'
                          r'seagrass|kelp|sugar_cane|bamboo|cactus|wheat|carrots|potatoes|beetroots|dirt_path|farmland|candle|pot$|'
                          r'head|skull|scaffolding|cobweb|dripstone|amethyst_cluster|bud$|chain|campfire|bell|anvil|grindstone|lectern|'
                          r'brewing_stand|cauldron|enchanting|stonecutter|end_rod|lightning_rod|frame|moss_carpet|petals|leaf_litter|'
                          r'redstone_block|observer|piston|copper_grate|pressure_plate|tripwire|string|repeater|comparator|redstone_wire|'
                          r'daylight|conduit|turtle_egg|sea_pickle|coral|pickle|azalea$|roots|sprouts|fungus|spore|hanging)')
CONDUCTOR_ALWAYS = re.compile(r'^minecraft:(soul_sand|mud)$')
def lid_blocks(name):
    if name is None: return False
    if CONDUCTOR_ALWAYS.match(name): return True
    return not NOT_BLOCKING.search(name)

# ------------------------------------------------------------------ colours (top-down)
COLOURS = [
    (r'water|bubble_column|seagrass|kelp', (52, 95, 190)), (r'lava', (230, 110, 20)),
    (r'leaves', (52, 110, 40)), (r'grass_block|short_grass|tall_grass|fern|moss', (100, 150, 60)),
    (r'dirt_path', (160, 130, 80)), (r'farmland', (110, 75, 45)), (r'^minecraft:(dirt|coarse_dirt|rooted_dirt|podzol|mud)$', (125, 90, 60)),
    (r'sand(?!stone)', (220, 210, 160)), (r'sandstone', (210, 195, 140)), (r'gravel', (135, 128, 125)), (r'snow|ice', (235, 240, 250)),
    (r'cobblestone|cobbled_deepslate', (118, 118, 118)), (r'stone_brick|bricks', (128, 128, 135)),
    (r'^minecraft:(stone|andesite|diorite|granite|tuff|deepslate|calcite|smooth_stone)$', (140, 140, 140)),
    (r'_log|_wood', (110, 85, 50)), (r'_planks|_slab|_stairs|_fence', (165, 130, 80)),
    (r'copper', (190, 110, 80)), (r'clay|terracotta', (160, 120, 100)), (r'flower|tulip|poppy|dandelion|daisy|allium|orchid|bluet|cornflower', (100, 150, 60)),
    (r'sapling', (70, 160, 60)), (r'pumpkin|melon', (200, 140, 30)), (r'obsidian', (30, 20, 45)),
]
COLOURS = [(re.compile(rx), c) for rx, c in COLOURS]
def colour(name):
    for rx, c in COLOURS:
        if rx.search(name): return c
    return (150, 150, 150)

MARK = {  # name -> (fill, outline)
    'container': ((255, 220, 0), (60, 40, 0)), 'container_blocked': ((230, 30, 30), (255, 255, 255)),
    'furnace': ((255, 140, 0), (40, 20, 0)), 'crafting_table': ((180, 120, 255), (30, 0, 60)),
    'composter': ((160, 255, 60), (20, 60, 0)), 'other_station': ((255, 255, 255), (0, 0, 0)),
    'torch': ((255, 250, 180), None), 'well': ((0, 230, 255), (0, 40, 60)), 'farm': ((0, 120, 0), (200, 255, 200)),
    'home': ((255, 0, 255), (255, 255, 255)),
}

def png(path, w, h, px):
    raw = b''.join(b'\x00' + bytes(px[y * w * 3:(y + 1) * w * 3]) for y in range(h))
    def chunk(t, d): c = struct.pack('>I', len(d)) + t + d; return c + struct.pack('>I', zlib.crc32(t + d) & 0xffffffff)
    with open(path, 'wb') as f:
        f.write(b'\x89PNG\r\n\x1a\n' + chunk(b'IHDR', struct.pack('>IIBBBBB', w, h, 8, 2, 0, 0, 0)) +
                chunk(b'IDAT', zlib.compress(raw, 9)) + chunk(b'IEND', b''))

def run_world(world, home, src, scratch, overlay, outdir):
    hx, hy, hz = home
    x0, x1, z0, z1 = hx - R, hx + R, hz - R, hz + R
    rdir = os.path.join(scratch, world, 'region'); edir = os.path.join(scratch, world, 'entities')
    os.makedirs(rdir, exist_ok=True); os.makedirs(edir, exist_ok=True)
    regs = {(cx >> 5, cz >> 5) for cx in range(x0 >> 4, (x1 >> 4) + 1) for cz in range(z0 >> 4, (z1 >> 4) + 1)}
    for rx, rz in regs:
        for sub, dst in (('region', rdir), ('entities', edir)):
            p = os.path.join(src, sub, 'r.%d.%d.mca' % (rx, rz))
            if os.path.exists(p): shutil.copyfile(p, os.path.join(dst, os.path.basename(p)))
    W = 2 * R + 1
    top = [[None] * W for _ in range(W)]; topy = [[-999] * W for _ in range(W)]
    pois = []
    stats = collections.Counter()
    for cx in range(x0 >> 4, (x1 >> 4) + 1):
        for cz in range(z0 >> 4, (z1 >> 4) + 1):
            ch, st = chunk_nbt(rdir, cx, cz)
            stats[st] += 1
            if ch is None: continue
            secs = sorted(ch.get('sections', []), key=lambda s: -s.get('Y', 0))
            colsleft = {(lx, lz) for lx in range(16) for lz in range(16) if x0 <= cx * 16 + lx <= x1 and z0 <= cz * 16 + lz <= z1}
            need_surface = set(colsleft)
            for sec in secs:
                bs = sec.get('block_states') or {}
                pal = [p.get('Name', '?') for p in (bs.get('palette') or [])]
                props = [p.get('Properties', {}) for p in (bs.get('palette') or [])]
                if not pal: continue
                y0 = sec['Y'] * 16
                single_air = len(pal) == 1 and pal[0] in AIR
                has_poi = any(POI.match(n) for n in pal)
                if single_air: continue
                if not has_poi and not need_surface: continue
                vals = section_values(bs)
                poi_idx = {i for i, n in enumerate(pal) if POI.match(n)}
                def at(lx, ly, lz):
                    return 0 if vals is None else vals[ly * 256 + lz * 16 + lx]
                for (lx, lz) in list(colsleft):
                    for ly in range(15, -1, -1):
                        v = at(lx, ly, lz); n = pal[v] if v < len(pal) else '?'
                        if (lx, lz) in need_surface and n not in AIR:
                            gx, gz = cx * 16 + lx - x0, cz * 16 + lz - z0
                            top[gz][gx] = n; topy[gz][gx] = y0 + ly
                            need_surface.discard((lx, lz))
                        if v in poi_idx:
                            pr = props[v] if v < len(props) else {}
                            pois.append({'n': n[10:], 'p': [cx * 16 + lx, y0 + ly, cz * 16 + lz], 'type': pr.get('type'), 'facing': pr.get('facing')})
                        if not has_poi and (lx, lz) not in need_surface: break
    # lid test needs the block above each chest
    blockcache = {}
    def block_at(x, y, z):
        cx, cz = x >> 4, z >> 4
        key = (cx, cz)
        if key not in blockcache:
            ch, _ = chunk_nbt(rdir, cx, cz)
            blockcache[key] = {s['Y']: s.get('block_states') for s in (ch or {}).get('sections', [])}
        bs = blockcache[key].get(y >> 4)
        if not bs: return None
        pal = [p.get('Name', '?') for p in (bs.get('palette') or [])]
        if len(pal) == 1: return pal[0]
        vals = section_values(bs)
        v = vals[(y & 15) * 256 + (z & 15) * 16 + (x & 15)]
        return pal[v] if v < len(pal) else '?'
    for p in pois:
        if p['n'] in ('chest', 'trapped_chest'):
            above = block_at(p['p'][0], p['p'][1] + 1, p['p'][2])
            p['above'] = above[10:] if above else None
            p['lid_blocked'] = lid_blocks(above)
    # entities
    ents = collections.Counter(); frames = []
    for cx in range(x0 >> 4, (x1 >> 4) + 1):
        for cz in range(z0 >> 4, (z1 >> 4) + 1):
            ch, st = chunk_nbt(edir, cx, cz)
            if ch is None: continue
            for e in ch.get('Entities', []) or []:
                pos = e.get('Pos') or [0, 0, 0]
                if x0 <= pos[0] <= x1 + 1 and z0 <= pos[2] <= z1 + 1:
                    ents[e.get('id', '?')] += 1
                    if 'frame' in e.get('id', ''): frames.append([round(v) for v in pos])
    # ---------------------------------------------------------------- render
    S = SCALE; Wp = W * S
    px = bytearray(Wp * Wp * 3)
    def put(gx, gz, c, inset=0):
        for yy in range(gz * S + inset, gz * S + S - inset):
            for xx in range(gx * S + inset, gx * S + S - inset):
                if 0 <= xx < Wp and 0 <= yy < Wp:
                    i = (yy * Wp + xx) * 3; px[i:i + 3] = bytes(c)
    def outline(gx, gz, c, size=1):
        ax0, az0 = gx * S - 1, gz * S - 1; ax1, az1 = gx * S + S * size, gz * S + S * size
        for xx in range(ax0, ax1 + 1):
            for yy in (az0, az1):
                if 0 <= xx < Wp and 0 <= yy < Wp: i = (yy * Wp + xx) * 3; px[i:i + 3] = bytes(c)
        for yy in range(az0, az1 + 1):
            for xx in (ax0, ax1):
                if 0 <= xx < Wp and 0 <= yy < Wp: i = (yy * Wp + xx) * 3; px[i:i + 3] = bytes(c)
    for gz in range(W):
        for gx in range(W):
            n = top[gz][gx]
            if n is None: put(gx, gz, (0, 0, 0)); continue
            c = colour(n)
            y = topy[gz][gx]; yn = topy[gz - 1][gx] if gz > 0 else y; yw = topy[gz][gx - 1] if gx > 0 else y
            d = (y - yn) + (y - yw)
            f = 1.0 + max(-0.35, min(0.35, 0.09 * d)) + 0.004 * (y - hy)
            put(gx, gz, tuple(max(0, min(255, int(v * f))) for v in c))
    # zone squares: Chebyshev 16 (no-dig zone / town) and 22 (outer domain); Euclid 44 circle (outer bound)
    def dot(xx, yy, c):
        if 0 <= xx < Wp and 0 <= yy < Wp: i = (yy * Wp + xx) * 3; px[i:i + 3] = bytes(c)
    cpx = R * S + S // 2
    for r, col in ((16, (255, 255, 255)), (22, (255, 230, 120))):
        for t in range(-r * S, r * S + 1):
            if (t // 3) % 2 == 0:
                for (xx, yy) in ((cpx + t, cpx - r * S), (cpx + t, cpx + r * S), (cpx - r * S, cpx + t), (cpx + r * S, cpx + t)):
                    dot(xx, yy, col)
    for a in range(0, 3600, 3):
        th = math.radians(a / 10)
        if (a // 30) % 2 == 0: dot(int(cpx + 44 * S * math.cos(th)), int(cpx + 44 * S * math.sin(th)), (255, 230, 120))
    def g(p): return p[0] - x0, p[2] - z0
    inbox = lambda p: x0 <= p[0] <= x1 and z0 <= p[2] <= z1
    for c in overlay.get('farm', []):
        if inbox(c): gx, gz = g(c); outline(gx, gz, MARK['farm'][1])
    for c in overlay.get('well_cells', []):
        if inbox(c): gx, gz = g(c); put(gx, gz, (0, 150, 190), 1)
    for p in pois:
        n = p['n']; gx, gz = g(p['p'])
        if n in ('torch', 'wall_torch', 'lantern', 'soul_torch'): put(gx, gz, MARK['torch'][0], 1)
    order = ['container', 'furnace', 'crafting_table', 'composter', 'other_station']
    def kind(n):
        if n in ('chest', 'trapped_chest', 'barrel'): return 'container'
        if n in ('furnace', 'blast_furnace', 'smoker'): return 'furnace'
        if n == 'crafting_table': return 'crafting_table'
        if n == 'composter': return 'composter'
        if n in ('anvil', 'smithing_table', 'stonecutter', 'grindstone', 'lectern', 'brewing_stand', 'bell', 'hopper', 'campfire') or n.endswith('_bed'): return 'other_station'
        return None
    near = lambda p: abs(p['p'][1] - hy) <= 12
    for p in pois:
        k = kind(p['n'])
        if not k or not near(p): continue
        if k == 'container' and p.get('lid_blocked'): k = 'container_blocked'
        gx, gz = g(p['p']); put(gx, gz, MARK[k][0]); outline(gx, gz, MARK[k][1])
    for c in overlay.get('well_caps', []):
        if inbox(c): gx, gz = g(c); outline(gx - 1, gz - 1, MARK['well'][0], 3)
    put(R, R, MARK['home'][0]); outline(R - 1, R - 1, MARK['home'][1], 3)
    os.makedirs(outdir, exist_ok=True)
    png(os.path.join(outdir, world + '.png'), Wp, Wp, px)
    # ---------------------------------------------------------------- summary
    counts = collections.Counter(); deep = collections.Counter()
    for p in pois:
        (counts if near(p) else deep)[p['n']] += 1
    surf = collections.Counter(n[10:] for row in top for n in row if n)
    heights = [topy[gz][gx] for gz in range(W) for gx in range(W) if top[gz][gx]]
    return {'world': world, 'home': home, 'box': [x0, z0, x1, z1], 'chunks': dict(stats),
            'near_counts': dict(counts.most_common()), 'deep_counts': dict(deep.most_common()),
            'surface_top': dict(surf.most_common(25)), 'entities': dict(ents.most_common()), 'frames': frames,
            'surface_y': {'min': min(heights), 'max': max(heights), 'home_y': hy} if heights else None,
            'pois': pois, 'top': [[(n[10:] if n else None) for n in row] for row in top], 'topy': topy}

def main():
    worlds = json.load(open(sys.argv[1])); overlays = json.load(open(sys.argv[2])); outdir = sys.argv[3]
    scratch = '/var/tmp/townplan-1020'
    res = {}
    for w, home in worlds.items():
        r = run_world(w, home, '/srv/block2/%s/world' % w, scratch, overlays.get(w, {}), outdir)
        json.dump(r, open(os.path.join(scratch, w + '.json'), 'w'))
        print(w, r['chunks'], 'near', {k: r['near_counts'].get(k, 0) for k in ('chest', 'barrel', 'furnace', 'crafting_table', 'composter', 'torch', 'dirt_path', 'farmland')},
              'ents', sum(r['entities'].values()), 'frames', len(r['frames']), flush=True)
        _region_cache.clear()

if __name__ == '__main__':
    main()
