"""es_bags.py: the newest bag snapshot per bot (ES, last 60 min): stations and containers carried."""
import json, sys, collections
from lib.estelemetry import EsEvents
ev = EsEvents.load('skill', since_minutes=60)
lat = ev.latest(require='bot.pos.x', fields=['bot.inventory', 'exp.pool', 'bot.pos'])
ITEMS = ['crafting_table', 'furnace', 'chest', 'barrel', 'composter', 'torch', 'leather', 'item_frame', 'oak_sign', 'stick', 'oak_planks', 'oak_log', 'cobblestone', 'iron_ingot', 'coal', 'copper_ingot']
per = collections.defaultdict(list)
held = collections.Counter(); carriers = collections.Counter(); slots = []
for bot, src in lat.items():
    inv = (src.get('bot') or {}).get('inventory') or {}
    slots.append(len(inv))
    for k in ITEMS:
        n = inv.get(k, 0) or 0
        held[k] += n
        if n: carriers[k] += 1
        per[k].append(n)
out = {'bots': len(lat), 'held': dict(held), 'carriers': dict(carriers),
       'max_one_bot': {k: max(v) if v else 0 for k, v in per.items()},
       'bags_ge_2_tables': sum(1 for n in per['crafting_table'] if n >= 2),
       'bags_ge_2_furnaces': sum(1 for n in per['furnace'] if n >= 2),
       'distinct_names_per_bag_median': sorted(slots)[len(slots) // 2] if slots else None,
       'caveats': list(ev.caveats)}
json.dump(out, sys.stdout, indent=1)
