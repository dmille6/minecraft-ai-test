from collections import Counter, defaultdict, deque

HOMES = {'board-a': (355,73,147), 'board-b': (-249,65,144), 'board-c': (355,73,147), 'board-d': (355,73,147),
         'hive-a': (355,73,147), 'hive-b': (355,73,147), 'hive-c': (-68,68,-68), 'hive-d': (68,65,68),

STONEFAM = set(COB) | {'stone', 'deepslate', 'andesite', 'diorite', 'granite', 'tuff', 'calcite', 'cobbled_deepslate'}
def pool_of(b): return '-'.join(b.split('-')[:2])
def dband(d):
    for lim in (8, 16, 24, 32, 48, 64, 96, 128, 256): 
        if d <= lim: return '<=%d' % lim
    return '>256'
def yband(dy):
    for lim in (2, -3, -8, -12, -20, -32, -48, -64, -96):
        if dy >= lim: return '>=%d' % lim
    return '<-96'
def srcclass(src):
    if src.startswith('mine'): return 'mine'
    if src.startswith('gather:'):
        a = src.split(':', 1)[1]
        if a in ('cobblestone', 'stone', 'cobbled_deepslate', 'deepslate'): return 'gather_stone'
        if a.endswith('_ore') or a in ('raw_iron', 'iron', 'coal', 'raw_copper'): return 'gather_ore'
        return 'gather_other'
    if src.startswith('goto') or src.startswith('explore') or src.startswith('surface') or src.startswith('return'): return 'transit:' + src.split(':')[0]
    if src.startswith('holder:') or src.startswith('reflex:'): return 'survival'
    if src == 'idle': return 'idle'
    return 'other:' + src.split(':')[0]

# r3 (Codex r2 P2-6): credit only the part of each call's descent that lies above y=16 (pro-rated by blocks dropped);
# a call that did not descend (drop 0) is credited only if it started above 16.
def above16(c):
    d = drop(c)
    if d <= 0: return c['cob'] if c['y0'] > 16 else 0
    return c['cob'] * max(0.0, c['y0'] - max(c['y1'], 16)) / d

    for dl, g in (('<=48', lambda c: c['d0'] <= 48), ('>48', lambda c: c['d0'] > 48)):
        xs = [c for c in sc if f(c) and g(c)]
        print('  %-12s %-5s calls %5d cobble %5d above16 %7.0f' % (lab, dl, len(xs), sum(c['cob'] for c in xs), sum(above16(c) for c in xs)))

# Ziggurat voxel model: tiers 2 high, setback S per side, base 48. Solid vs hollow (26-adjacent skin) + support check.
import itertools, sys
def model(B=48, S=3, H=2, stair_w=3):
    tiers = []
    L = B; i = 0
    while L >= 4:
        tiers.append((i, L)); i += 1; L -= 2 * S
    solid = set()
    for i, L in tiers:
        o = (B - L) // 2
        for y in range(i * H, i * H + H):
            for x in range(o, o + L):
                for z in range(o, o + L):
                    solid.add((x, y, z))
    top = len(tiers) * H
    def outside(c):
        return c not in solid
    skin = set()
    for c in solid:
        x, y, z = c
        for dx, dy, dz in itertools.product((-1, 0, 1), repeat=3):
            if (dx, dy, dz) == (0, 0, 0): continue
            n = (x + dx, y + dy, z + dz)
            if n[1] < 0: continue          # ground below is not air
            if n not in solid:
                # interior air does not exist in the solid model, so any non-solid neighbour above ground is outside
                skin.add(c); break
    # support: every skin cell has a skin cell below, or y==0 (ground), or a horizontal skin neighbour (bridge risk counted)
    nosup = [c for c in skin if c[1] > 0 and (c[0], c[1] - 1, c[2]) not in skin]
    per_tier = []
    for i, L in tiers:
        ys = range(i * H, i * H + H)
        per_tier.append((i, L, sum(1 for c in skin if c[1] in ys), sum(1 for c in solid if c[1] in ys)))
    # grand stair: on the south face, stair_w wide, one full block step per tier at the foot of each tier wall (i>=1)
    # tier i wall rises 2 from terrace of tier i-1: steps needed = 1 block at terrace level (+1), per width; plus ground->tier0: 1 step
    stairs = stair_w * len(tiers)
    return tiers, len(solid), len(skin), len(nosup), per_tier, stairs, top
for S in (2, 3, 4):
    tiers, ns, nk, nos, pt, st, top = model(S=S)
    print('S=%d tiers=%d height=%d solid=%d hollow_skin=%d unsupported_skin=%d stairs=%d' % (S, len(tiers), top, ns, nk, nos, st))
    for row in pt: print('   tier %d side %d: hollow %d solid %d' % row)
