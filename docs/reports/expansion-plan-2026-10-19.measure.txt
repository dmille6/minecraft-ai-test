################ SCRIPT h1.py
"""home diagnosis, read only. 7 d of `home` skill rows from ES, each joined to the bot's PREVIOUS row (its start position)."""
import sys, json, math, re, collections, datetime as dt
sys.path.insert(0, '/home/mike/estel/scripts')
from lib.estelemetry import EsEvents

SINCE = '2026-10-12T15:00:00Z'
HOMES = {'board-a': (355, 73, 147), 'board-b': (-249, 65, 144), 'board-c': (355, 73, 147), 'board-d': (355, 73, 147),
         'hive-a': (355, 73, 147), 'hive-b': (355, 73, 147), 'hive-c': (-68, 68, -68), 'hive-d': (68, 65, 68),
         'placebo-a': (249, 75, -144), 'placebo-b': (282, 65, -388), 'placebo-c': (355, 73, 147), 'placebo-d': (177, 75, 73)}
ISO = (355, 73, 147)

ev = EsEvents.load('skill', since=SINCE)
print('window', ev.since, ev.until, 'docs', ev.total(), 'bots', len(ev.bots_present()))
print('caveats', ev.caveats)
rows = list(ev.rows('home', fields=['@timestamp', 'exp.pool', 'bot.name', 'bot.pos', 'bot.health', 'skill', 'trigger'], max_rows=20000))
print('home rows', len(rows))

def prev_row(bot, ts):
    body = {'size': 1, '_source': ['@timestamp', 'bot.pos', 'skill.name', 'skill.status', 'trigger'],
            'query': {'bool': {'filter': [{'term': {'bot.name': bot}},
                                          {'range': {'@timestamp': {'lt': ts, 'gte': (dt.datetime.fromisoformat(ts.replace('Z', '+00:00')) - dt.timedelta(minutes=15)).isoformat()}}}]}},
            'sort': [{'@timestamp': 'desc'}]}
    h = ev._search(body)['hits']['hits']
    return h[0]['_source'] if h else None

out = []
for i, r in enumerate(rows):
    pool = r['exp']['pool']
    home = HOMES.get(pool, ISO)
    p = r.get('bot', {}).get('pos') or {}
    sk = r.get('skill', {})
    pr = prev_row(r['bot']['name'], r['@timestamp'])
    sp = (pr or {}).get('bot', {}).get('pos')
    def d(pp):
        if not pp or pp.get('x') is None: return None
        return math.hypot(pp['x'] - home[0], pp['z'] - home[2])
    out.append({'t': r['@timestamp'], 'pool': pool, 'bot': r['bot']['name'], 'shared': pool in HOMES,
                'status': sk.get('status'), 'fc': sk.get('fail_class'), 'detail': (sk.get('detail') or '')[:240],
                'trigger': r.get('trigger'), 'dur': sk.get('duration_ms'), 'moved': sk.get('distance_moved'),
                'end_d': d(p), 'end_y': p.get('y'), 'start_d': d(sp), 'start_y': (sp or {}).get('y'), 'home_y': home[1],
                'prev': (pr or {}).get('skill', {}).get('name'), 'prev_status': (pr or {}).get('skill', {}).get('status'),
                'prev_trig': (pr or {}).get('trigger'), 'health': r.get('bot', {}).get('health')})
json.dump(out, open('/tmp/exp_home_rows.json', 'w'))
print('written', len(out))

        if d <= b: return b

samples = []
