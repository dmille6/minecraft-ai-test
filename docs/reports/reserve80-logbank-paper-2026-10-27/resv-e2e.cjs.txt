    need((ta.bag.coal || 0) === 3, `the bag keeps its OWN 3 coal (${ta.bag.coal || 0})`)
    need((ta.bag.iron_ingot || 0) === 2, `the bag keeps its OWN 2 iron_ingot (${ta.bag.iron_ingot || 0})`)
    need(dTown(tb, ta, 'coal') === 0, `the town's coal untouched (${dTown(tb, ta, 'coal')})`)
    need(dTown(tb, ta, 'iron_ingot') === 0, `the town's iron untouched (${dTown(tb, ta, 'iron_ingot')})`)
    const judged = Object.fromEntries(Object.entries(ta.ground).filter(([n]) => JUDGED.has(n) || !AMBIENT.test(n)))
    need(!Object.keys(judged).length, `nothing the job or the bag holds on the ground (${diff({}, judged)})`)
    need(!rows.some(r => /verdict=(loss|forbidden|overburn)/.test(r.detail)), 'no row with a loss, forbidden or overburn verdict')
    for (const t of ['iron_pickaxe', 'iron_axe', 'iron_shovel']) need((ta.bag[t] || 0) === 1, `the bag keeps its ${t} (${ta.bag[t] || 0}) -- B3: never deposited, never an input (Codex r2 P2-10)`)
  }
  const furnaceEmpty = k => !Object.keys(ta.furnace[k] || {}).length && !a.lit[k]
  out.ambient = diff({}, Object.fromEntries(Object.entries(ta.ground).filter(([n]) => !JUDGED.has(n) && AMBIENT.test(n))))
  if (!CANDIDATE) {
    for (const n of ['cobblestone', 'oak_log', 'stone_bricks', 'copper_ingot', 'cut_copper', 'copper_door', 'torch', 'charcoal', 'glass_pane', 'sand', 'coal', 'iron_ingot']) need(dTown(tb, ta, n) === 0, `the base leaves the town's ${n} as it was (${dTown(tb, ta, n)})`)
    need(!jobs.length, 'the base writes no _make_material row')
    out.pass = !out.why.length
    out.verdict = out.pass ? 'BASE_SHAPE' : 'BASE_UNEXPECTED'
    return out
  }
  need(rows.some(r => r.name === '_mats_mode'), 'the licence row _mats_mode was written (the build is live)')
  if (/^resv/.test(scene)) {
    const sp = SPEC[scene], lowered = !/off$|norec$/.test(scene) || scene === 'resvlogoff'
    const draws = rows.filter(r => r.name === '_reserve_draw')
    const modes = rows.filter(r => r.name === '_reserve_mode')
    out.reserve = { draws: draws.map(r => r.detail), modes: modes.map(r => r.detail), made: made.map(r => r.detail.slice(0, 300)),
      townEnd: { cobblestone: town(ta, 'cobblestone'), oak_log: town(ta, 'oak_log'), stone_bricks: town(ta, 'stone_bricks') },
      bagLogs: [tb.bag.oak_log || 0, ta.bag.oak_log || 0], banked: rows.filter(r => r.name === '_town_deposit').map(r => r.detail.slice(0, 200)) }
    // r2: the logbank scenes START under the lowered log floor (100 < 103) by construction; the floor is judged on the rest
    if (scene !== 'resvlogoff') need(town(ta, 'cobblestone') >= 52 && town(ta, 'oak_log') >= 103, `never under the lowered floor (cobble ${town(ta, 'cobblestone')}, logs ${town(ta, 'oak_log')})`)
    if (scene === 'resv' || scene === 'resvlog') {
      need(modes.some(r => /active=1 why=active/.test(r.detail)), 'a _reserve_mode active=1 row')
      need(made.some(r => /kind=bricks/.test(r.detail) && /(?:^| )reserve=project/.test(r.detail)), 'a bricks order made parts under reserve=project')
      need(draws.some(r => /family=cobblestone/.test(r.detail) && /floor_effective=52/.test(r.detail)), 'a _reserve_draw row for cobblestone at floor 52')
      need(dTown(tb, ta, 'stone_bricks') + (ta.bag.stone_bricks || 0) >= 12 || rows.some(r => r.name === '_build_bag'), `stone_bricks made (town ${dTown(tb, ta, 'stone_bricks')})`)
    }
    // r2: the server's chest gain equals the row's banked count and the row kept 16 (the bag may then spend logs on the
    // workshop's own station crafts -- _build_bag books them -- so the END bag is not 16 by construction)
    const banked = rows.map(r => r.name === '_town_deposit' && /logs banked (\d+) kept (\d+)/.exec(r.detail)).find(Boolean)
    out.reserve.logsBankedRow = banked ? [Number(banked[1]), Number(banked[2])] : null
    if (scene === 'resvlog') {
      need(!!banked && Number(banked[2]) === 16 && Number(banked[1]) >= 50 && dTown(tb, ta, 'oak_log') === Number(banked[1]) - 2, `logbank: the row kept 16 and the server's town gained the banked logs less the 2 bricks fuel (row ${banked?.[0]}, town d ${dTown(tb, ta, 'oak_log')})`)
      const tDep = rows.find(r => r.name === '_town_deposit')?.ts ?? Infinity, tDraw = rows.find(r => r.name === '_reserve_draw' && /family=logs/.test(r.detail))?.ts ?? -1
      need(tDep < tDraw, 'the logs were banked BEFORE the lowered-floor log draw')
    }
    if (scene === 'resvlogoff') {
      need(!banked || Number(banked[1]) === 0, `LOGBANK=off: no log banked (${banked?.[0]})`)
      need(dTown(tb, ta, 'oak_log') === 0 && !made.some(r => /kind=bricks/.test(r.detail)), `no bricks: the town's 100 logs stay under the lowered 103 (d logs ${dTown(tb, ta, 'oak_log')})`)
      need(skipSays(rows, /kinds=\S*bricks:inputs/), 'the skip says bricks:inputs at a known town')
    }
    if (scene === 'resvoff' || scene === 'resvnorec') {
      need(!made.length, 'no part made (the full floor: cobble 200 < 256)')
      need(!draws.length, 'no _reserve_draw row')
      need(dTown(tb, ta, 'cobblestone') === 0, `the town's cobble unchanged (${dTown(tb, ta, 'cobblestone')})`)
      need(skipSays(rows, /kinds=\S*bricks:inputs/), 'the skip says bricks:inputs at a known town')
    }
    void sp; void lowered
    out.pass = !out.why.length
    out.verdict = out.pass ? 'CANDIDATE_PASS' : 'FAIL'
    return out
  }
  const bricksMade = (cob, logs) => {
    need(dTown(tb, ta, 'stone_bricks') === 12, `the town's stone_bricks +12 (${dTown(tb, ta, 'stone_bricks')})`)
    need(dTown(tb, ta, 'cobblestone') === -cob, `the town's cobblestone -${cob} (${dTown(tb, ta, 'cobblestone')})`)
    need(dTown(tb, ta, 'oak_log') === -logs, `the town's oak_log -${logs} (${dTown(tb, ta, 'oak_log')})`)
    need(!(ta.bag.stone > 0) && !(ta.bag.stone_bricks > 0) && !(ta.bag.oak_planks > 0) && !(ta.bag.cobblestone > 0) && !(ta.bag.oak_log > 0), `the bag ends as it began (${diff(tb.bag, ta.bag)})`)
  }
  if (scene === 'bricks' || scene === 'res_one') {
    need(made.length === 1 && /kind=bricks/.test(made[0].detail), `exactly one bricks order made parts (${made.length})`)
    bricksMade(12, 2)
    need(furnaceEmpty('F1'), 'the furnace is left empty and cold')
    if (scene === 'res_one') need(town(ta, 'cobblestone') === 256 && town(ta, 'oak_log') === 512, `the town ends at the reserve (${town(ta, 'cobblestone')} cobble, ${town(ta, 'oak_log')} logs)`)
  }
  if (scene === 'copper') {
    need(dTown(tb, ta, 'cut_copper') === 4, `cut_copper +4 (${dTown(tb, ta, 'cut_copper')})`)
    need(dTown(tb, ta, 'copper_ingot') === -36, `copper_ingot -36 (${dTown(tb, ta, 'copper_ingot')})`)
    need(!(ta.bag.copper_ingot > 0) && !(ta.bag.copper_block > 0) && !(ta.bag.cut_copper > 0), `the bag holds no copper after (${diff(tb.bag, ta.bag)})`)
  }
  if (scene === 'furn') {
    bricksMade(12, 2)
    need((ta.furnace.F1.coal || 0) === 3 && Object.keys(ta.furnace.F1).length === 1, `F1's 3 coal untouched (${JSON.stringify(ta.furnace.F1)})`)
    need((ta.furnace.F2.iron_ingot || 0) === 4 && Object.keys(ta.furnace.F2).length === 1, `F2's 4 iron untouched (${JSON.stringify(ta.furnace.F2)})`)
    need((ta.furnace.F4.stone || 0) === 3 && Object.keys(ta.furnace.F4).length === 1, `F4's 3 stone untouched -- another session's output (${JSON.stringify(ta.furnace.F4)})`)
    need(furnaceEmpty('F3'), 'F3 (the clean one) left empty and cold')
    need(rows.some(r => r.name === '_furnace_session' && /by=make_material/.test(r.detail) && r.detail.includes(`furnace=${F3.x},${F3.y},${F3.z}`)), 'the job\'s own session ran in F3 (its _furnace_session row names it)')
    // a session row on F1/F2/F4 is the job's QUALIFICATION PROBE (lease, open, refuse untouched: Codex r3 P2-7) -- correct; the
    // foreign furnaces are judged by their CONTENTS above
  }
  if (scene === 'hot' || scene === 'res_at') {
    need(!made.length, 'no part made')
    for (const n of ['cobblestone', 'oak_log', 'stone_bricks']) need(dTown(tb, ta, n) === 0, `the town's ${n} unchanged (${dTown(tb, ta, n)})`)
    if (scene === 'hot') need(!Object.keys(ta.furnace.F1).length, `nothing loaded into the hot furnace (${JSON.stringify(ta.furnace.F1)})`)
    const want = scene === 'hot' ? 'bricks:no_furnace' : 'bricks:inputs'
    need(skipSays(rows, new RegExp(`kinds=\\S*${want}`)), `the skip row names ${want} at a coverage >= 0.8 (the town was known)`)
  }
  if (scene === 'torch') {
    need(dTown(tb, ta, 'torch') === 24, `torch +24 in town (${dTown(tb, ta, 'torch')})`)
    need(dTown(tb, ta, 'oak_log') === -8, `oak_log -8 (${dTown(tb, ta, 'oak_log')})`)
    need(!(ta.bag.charcoal > 0) && !dTown(tb, ta, 'charcoal') && !Object.keys(ta.furnace.F1 || {}).length, `no charcoal left in the bag, the town or the furnace (bag ${ta.bag.charcoal || 0})`)
    need((ta.bag.torch || 0) === 0, `no made torch left in the bag (${ta.bag.torch || 0})`)
    need((ta.bag.stick || 0) <= 2 && !(ta.bag.oak_planks > 0) && !(ta.bag.oak_log > 0), `at most the 2 spare sticks left (${diff(tb.bag, ta.bag)})`)
  }
  if (scene === 'panes') {
    need(dTown(tb, ta, 'glass_pane') === 16, `glass_pane +16 (${dTown(tb, ta, 'glass_pane')})`)
    need(dTown(tb, ta, 'sand') === -6, `sand -6 (${dTown(tb, ta, 'sand')})`)
    need(dTown(tb, ta, 'oak_log') === -1, `oak_log -1 (${dTown(tb, ta, 'oak_log')})`)
    need(!(ta.bag.glass > 0) && !dTown(tb, ta, 'glass') && !Object.keys(ta.furnace.F1 || {}).length, `no glass left in the bag, the town or the furnace (bag ${ta.bag.glass || 0})`)
  }
  if (scene === 'door') {
    need(dTown(tb, ta, 'copper_door') === 3, `copper_door +3 (${dTown(tb, ta, 'copper_door')})`)
    need(dTown(tb, ta, 'copper_ingot') === -6, `copper_ingot -6 (${dTown(tb, ta, 'copper_ingot')})`)
    need(!(ta.bag.copper_ingot > 0) && !(ta.bag.copper_door > 0), `the bag holds no copper after (${diff(tb.bag, ta.bag)})`)
  }
  if (scene === 'kill_w') {
    need(rows.some(r => r.name === '_mats_return' && /stop=closed/.test(r.detail) && /batch=0/.test(r.detail)), 'the recovery closed the note with nothing in flight (batch=0)')
    need(dTown(tb, ta, 'cobblestone') + (ta.bag.cobblestone || 0) === 0, `the 12 cobble are in the bag or back in town (town ${dTown(tb, ta, 'cobblestone')}, bag ${ta.bag.cobblestone || 0})`)
    need(!dTown(tb, ta, 'stone_bricks') && !(ta.bag.stone > 0), 'nothing was smelted')
  }
  if (scene === 'kill') {
    need(rows.some(r => r.name === '_mats_return' && /drained:[1-9]/.test(r.detail)), 'mats_return drained the note\'s furnace')
    const acc = dTown(tb, ta, 'stone_bricks') + (ta.bag.stone || 0) + (ta.bag.cobblestone || 0)
    need(acc === 12 && !Object.keys(ta.furnace.F1 || {}).length, `the 12 cobble taken are accounted: town bricks ${dTown(tb, ta, 'stone_bricks')} + bag stone ${ta.bag.stone || 0} + bag cobble ${ta.bag.cobblestone || 0} = ${acc}`)
    need((ta.bag.stone || 0) <= 3, `at most a 1-3 stone remainder (${ta.bag.stone || 0})`)
    need(rows.some(r => r.name === '_mats_return' && /stop=closed/.test(r.detail)), 'the recovery closed the note')
    need(dTown(tb, ta, 'cobblestone') === -12, `the town's cobblestone -12 (${dTown(tb, ta, 'cobblestone')})`)
    need(furnaceEmpty('F1'), 'the furnace is left empty and cold')
  }
  untouched()
  out.pass = !out.why.length && done
  if (!done) out.why.push('the scene did not reach its completion condition inside the window')
  out.verdict = out.pass ? 'CANDIDATE_PASS' : 'FAIL'
  return out
}

async function runScene (scene) {
  const spec = SPEC[scene]
  if (!spec) throw new Error(`unknown scene ${scene}`)
  const tag = `${scene}-${CANDIDATE ? 'cand' : 'base'}-${SERVER}-${new Date().toISOString().replace(/[-:]/g, '').slice(9, 15)}`
  const logRel = `./sandbox/log/mats-e2e/${tag}`
  const skillLog = `${R}/sandbox/log/mats-e2e/${tag}/skill-${NAME}.jsonl`
  const llmLog = `${R}/sandbox/log/mats-e2e/${tag}/llm-${NAME}.jsonl`
  const botOut = `${OUTDIR}/bot-${tag}.out`; const trace = `${OUTDIR}/trace-${tag}.jsonl`
  const envRel = ENV_REL
  let env = fs.readFileSync(`${R}/sandbox/sandbox-bot-scripted.env`, 'utf8')
  const set = (key, v) => { env = new RegExp(`^${key}=`, 'm').test(env) ? env.replace(new RegExp(`^${key}=.*$`, 'm'), `${key}=${v}`) : env + `\n${key}=${v}` }
  set('BOT_NAME', NAME); set('LOG_DIR', logRel); set('STATE_DIR', `./sandbox/state/mats-${tag}`); set('MEMORY_POOL', `sbxmt-${tag}`)
  set('MEMORY_SCOPE', 'shared'); set('MINECRAFT_PORT', PORTS[SERVER]); set('STUCK_SECONDS', 20); set('PLANT_ENABLED', 0); set('PLINTH_FIRST', 0)
  set('OLLAMA_BASE_URL', `http://127.0.0.1:${BRAIN_PORT}`); set('OLLAMA_BASE_URLS', `http://127.0.0.1:${BRAIN_PORT}`); set('MAX_CONSECUTIVE_FAILURES', 50)
  set('MATS_COOLDOWN_MS', 60000)              // sandbox pacing: one order a minute (the fleet runs 10 min)
  for (const [k, v] of Object.entries(spec.env ?? {})) set(k, v)
  for (const ax of ['X', 'Y', 'Z']) { set(`HOME_${ax}`, HOME[ax.toLowerCase()]); set(`BOARD_${ax}`, HOME[ax.toLowerCase()]) }
  fs.writeFileSync(`${R}/${envRel}`, env + '\n')
  brainLog = `${OUTDIR}/brain-${tag}.log`
  if (spec.seed) {
    const poolDir = `${R}/sandbox/state/_pool-sbxmt-${tag}`
    fs.mkdirSync(poolDir, { recursive: true })
    for (const n of ['project-760_120_760.shop.g1.json', 'townplan-760_120_760.g1.json']) fs.copyFileSync(path.join(SEED_POOL, n), path.join(poolDir, n))
  }
  await startBot(envRel, botOut, trace)
  rcon(`gamemode survival ${NAME}`, `tp ${NAME} ${STAND.x} ${STAND.y} ${STAND.z}`, `effect clear ${NAME}`, 'difficulty peaceful')
  await sleep(1500)
  const built = [...rcon(...arenaCmds()), ...rcon(...bagCmds(spec)), ...rcon(...sceneCmds(spec))]
  const bad = built.filter(x => /not loaded|Cannot place|Unknown|Incorrect|Expected/i.test(x.reply) && !/^(kill|forceload) /.test(x.cmd))
  if (bad.length) { await stopBot(); throw new Error('arena not built: ' + bad.map(x => x.cmd + ' => ' + x.reply).join('; ')) }
  await sleep(1500)
  const before = snapshot(spec)
  {
    const tb0 = totals(before)
    const want = list => list.reduce((m, [id, n]) => ({ ...m, [id]: (m[id] || 0) + n }), {})
    const same = (x, y) => JSON.stringify(Object.entries(x).sort()) === JSON.stringify(Object.entries(y).sort())
    const bad0 = [!same(tb0.a, want(spec.chestA)) && `chest A ${JSON.stringify(tb0.a)}`, !same(tb0.b, want(spec.chestB)) && `chest B ${JSON.stringify(tb0.b)}`,
      !same(tb0.bag, want(bagOf(spec))) && `bag ${JSON.stringify(tb0.bag)}`].filter(Boolean)
    if (scene === 'hot' && !before.lit.F1) bad0.push('the hot furnace does not read lit=true')
    if (bad0.length) { await stopBot(); throw new Error('the scene is not as specified before the measurement: ' + bad0.join('; ')) }
  }
  const t0 = Date.now()
  let killedAt = null
  if (scene === 'kill' && CANDIDATE) {
    // MAT-KILL: SIGKILL the bot while its batch burns (the furnace lit), then start it again on the same state and logs
    const lit = await waitFor(() => { try { return /Test passed/.test(rcon(`execute if block ${at(F1)} minecraft:furnace[lit=true]`)[0].reply) } catch { return false } }, WINDOW_MS, 2000)
    if (!lit) { await stopBot(); throw new Error('the kill scene never saw its furnace lit') }
    await sleep(4000)
    killedAt = Date.now()
    await stopBot('SIGKILL')
    await sleep(15000)
    await startBot(envRel, botOut, trace)
    rcon(`tp ${NAME} ${STAND.x} ${STAND.y} ${STAND.z}`)
  }
  if (scene === 'kill_w' && CANDIDATE) {
    // `clear <player> <item> 0` counts without removing anything (the server's own reply)
    const held = () => { try { return Number((/Found (\d+) matching/.exec(rcon(`clear ${NAME} minecraft:cobblestone 0`)[0].reply) || [])[1] || 0) } catch { return 0 } }
    const got = await waitFor(() => held() >= 12, WINDOW_MS, 500)
    if (!got) { await stopBot(); throw new Error('the kill_w scene never saw the 12 cobble in the bag') }
    killedAt = Date.now()
    await stopBot('SIGKILL')
    await sleep(15000)
    await startBot(envRel, botOut, trace)
    rcon(`tp ${NAME} ${STAND.x} ${STAND.y} ${STAND.z}`)
  }
  // a recovery waits out the burn and the dead process's lease (mats_return yields 90 s per wait): the kill scenes get 10 min
  const winMs = /^resv/.test(scene) && /off$|norec$/.test(scene) && scene !== 'resvlogoff' ? 300000 : CANDIDATE ? (/^kill/.test(scene) ? Math.max(WINDOW_MS, 600000) : WINDOW_MS) : BASE_WINDOW_MS
  const done = await waitFor(() => DONE(scene, skillRows(skillLog).filter(r => r.ts >= t0 - 5000)), winMs, 1000)
  if (done) await sleep(8000)
  const after = snapshot(spec)
  const rows = skillRows(skillLog)
  const sceneRows = rows.filter(r => r.ts >= t0 - 5000 || r.name === '_mats_mode')
  const dec = decisions(llmLog).filter(r => r.ts >= t0 - 5000)
  await stopBot()
  await sleep(1000)
  try { fs.unlinkSync(`${R}/${envRel}`) } catch {}
  const tb = totals(before), ta = totals(after)
  const v = verdict(scene, tb, ta, sceneRows, after, !!done)
  const r = { reserve: v.reserve, scene, tag, candidate: CANDIDATE, server: SERVER, killedAt,
    sha: (() => { try { return execFileSync('git', ['-C', BOT_ROOT, 'rev-parse', 'HEAD'], { encoding: 'utf8' }).trim() } catch { return '?' } })(),
    windowMs: Date.now() - t0, done: !!done, verdict: v, before: { ...before, totals: tb }, after: { ...after, totals: ta },
    d: { bag: diff(tb.bag, ta.bag), chestA: diff(tb.a, ta.a), chestB: diff(tb.b, ta.b), ground: diff(tb.ground, ta.ground),
      furnace: Object.fromEntries(Object.keys(spec.furnaces).map(k => [k, diff(tb.furnace[k], ta.furnace[k])])) },
    decisions: dec.map(d => `${d.order ? 'ORDER' : 'model'} ${d.skill} ${d.admitted ? d.status : 'REFUSED ' + d.err}`), rows: sceneRows }
  fs.appendFileSync(RES, JSON.stringify(r) + '\n')
  console.log(`${scene.padEnd(8)} ${v.verdict} done=${r.done} d(bag){${r.d.bag}} d(A){${r.d.chestA}} d(B){${r.d.chestB}} d(furnaces)${JSON.stringify(r.d.furnace)} ground{${r.d.ground}} ambient{${v.ambient}} lit=${JSON.stringify(after.lit)}` +
    (v.why.length ? `\n   why: ${v.why.join('; ')}` : ''))
  console.log('   decisions: ' + r.decisions.filter(d => /ORDER/.test(d)).slice(0, 14).join(' | '))
  console.log('   rows: ' + sceneRows.filter(x => x.name !== '_mats_mode').map(x => `${x.name}:${x.status}:${x.detail.slice(0, 260)}`).join(' || '))
  return v.pass
}

let arenaOwned = false
async function main () {
  sbx.requireSandboxLock(SERVER, 'reserve80')
  const pl = rcon('list')[0].reply; if (!/There are 0 of/.test(pl)) throw new Error('sandbox not empty: ' + pl)
  fs.mkdirSync(OUTDIR, { recursive: true })
  arenaOwned = true
  await new Promise((resolve, reject) => { brain.once('error', reject); brain.listen(BRAIN_PORT, '127.0.0.1', resolve) })
  console.log(`root=${BOT_ROOT} build=${CANDIDATE ? 'CANDIDATE' : 'BASE'} server=${SERVER} scenes=${SCENES_S}`)
  let ok = true
  for (const s of SCENES_S.split(',')) {
    try { ok = (await runScene(s)) && ok } catch (e) { ok = false; console.log(`${s} SCENE FAILED: ${e.message}`); await stopBot().catch(() => {}) }
  }
  return ok
}
async function cleanup () {
  if (arenaOwned) await stopBot().catch(() => {})
  try { brain.close() } catch {}
  try { fs.unlinkSync(`${R}/${ENV_REL}`) } catch {}
  if (arenaOwned) { try { rcon(`kill @e[type=!player,x=${HOME.x},y=${HOME.y},z=${HOME.z},distance=..20]`, 'fill 746 120 746 774 140 774 minecraft:air', 'forceload remove 744 744 776 776') } catch {} }
}
let cleaning = false
const finish = code => { if (cleaning) return; cleaning = true; cleanup().then(() => process.exit(code)) }
main().then(ok => finish(ok ? 0 : 2), e => { console.error('FATAL', e.message); finish(1) })
process.on('SIGINT', () => finish(1))
process.on('SIGTERM', () => finish(1))
process.on('SIGHUP', () => finish(1))
