import shopread as SR  # noqa: E402
pool = sys.argv[1]
files = []
for f in glob.glob(os.path.join(pool, 'project-*.shop.g*.json')):
    m = re.search(r'\.g(\d+)\.json$', f)
    files.append((int(m.group(1)), open(f, 'rb').read()))
rec = SR.record_head(files)
since = dt.datetime.fromtimestamp(rec['completeAt'] / 1000, dt.timezone.utc)
svs = [json.load(open(p)) for p in sys.argv[2:]]
for p, s in zip(sys.argv[2:], svs):
    print(os.path.basename(p), s.get('at'), 'f1', s.get('f1'), 'valid', s.get('valid'), 'gen', s.get('record_gen'), 'sha ok', s.get('record_sha') == rec['_sha'])
print('completion', since.isoformat(), '->', SR.conformance(svs, rec, since))
