#!/usr/bin/env python3
# negcmds.py <record.json> remove|swap -- the RCON commands for the conformance NEGATIVE on the sandbox (sandbox worlds only):
#   remove: both furnaces, the table and all four windows -> air (7 cells: the F1 0.979 case)
#   swap:   the first 4 `base` cells -> cobblestone (4 wrong names: F1 0.976 on their own)
# then `save-all flush` so the region files hold the change.
import json, sys
rec = json.load(open(sys.argv[1]))
mode = sys.argv[2]
cmds = []
if mode == 'remove':
    for c in rec['cells']:
        if c['role'] in ('furnace', 'table', 'window'):
            cmds.append('setblock %d %d %d minecraft:air' % tuple(c['at']))
elif mode == 'swap':
    for c in [c for c in rec['cells'] if c['role'] == 'base'][:4]:
        cmds.append('setblock %d %d %d minecraft:cobblestone' % tuple(c['at']))
cmds.append('save-all flush')
print('\n'.join(cmds))
