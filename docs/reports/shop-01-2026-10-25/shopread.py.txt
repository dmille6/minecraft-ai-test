def tidy_ok(rec, cell):
    """zigtidy-01: is a `kind=tidy` clear inside its record's tidy space? -> bool. Pure."""
    cells = rec.get('cells') or []
    box = rec.get('box')
    if not cells or not isinstance(box, list) or len(box) != 4 or not isinstance(rec.get('baseY'), int):
        return False
    if ck(cell) in {ck(c['at']) for c in cells}:
        return False
    top = max(c['at'][1] for c in cells)
    x, y, z = cell
    return box[0] - 1 <= x <= box[2] + 1 and box[1] - 1 <= z <= box[3] + 1 and rec['baseY'] <= y <= top + 2


def tidy_clear_ok(chain, cell, kind, t_ms, guard):
    """tidyread-01 (11-03): is a `project=tidy` clear one the tidy record planned? -> bool. Pure.
    chain: every generation of the tidy record's chain, root first (record_chain). A stage is one generation (tidy.mjs siteStage),
    so a cut planned in stage 0 stays legitimate after the head moves on to stage 1. Planned = a cell of any generation's `cut`
    list (kind=tidy: the pillar and grading cuts) or `clear` list (the road's leaves and trunks); kind=foreign: a cell of any
    generation's `cells` (the fill cells). guard: {cell key: ms it was first reserved} -- every OTHER project's record cell and every
    expedition marker block. A cell guarded at or before the clear (t_ms) is never the tidy's, planned or not (buildTidy's own dig
    guard: the projects' cells + the markers); a guard that came later (a house sited on a dug pillar) does not re-judge it."""
    k = ck(cell)
    if tidy_guarded(cell, t_ms, guard):
        return False
    lists = ('cells',) if kind == 'foreign' else ('cut', 'clear')
    return any(k == ck(c['at']) for rec in chain or [] for name in lists for c in rec.get(name) or [] if isinstance(c, dict) and c.get('at'))


def gather_ok(chain, cell, block, t_ms, guard, park_box):
    """gkhelpers-01: is a `project=tidy op=clear kind=gather` (the park gather, bots/src/skills.mjs parkGather) a plant the gather may
    take? -> (bool, why). Pure. A GATHER_BLOCKS block; never guarded (another project's record cell or a marker when it was dug:
    tidy_guarded), never inside the park's box + 1 (park_box [x0, z0, x1, z1]: the land plan's), never a cell of any generation of the
    tidy chain (a cut or a clear-list leaf may be taken). The protected zones are G3's."""
    if block not in GATHER_BLOCKS:
        return False, ''
    guarded = tidy_guarded(cell, t_ms, guard)
    if guarded:
        return False, ' guarded'
    if isinstance(park_box, list) and len(park_box) == 4 and park_box[0] - 1 <= cell[0] <= park_box[2] + 1 and park_box[1] - 1 <= cell[2] <= park_box[3] + 1:
        return False, ' in_park'
    # a FILL cell of any generation (a park planting is one too) -- never the cut or clear lists: a head-room leaf the tidy itself
    # would clear is a plant the gather may take (Claude r1 BLOCKER 3)
    k = ck(cell)
    if any(k == ck(c['at']) for rec in chain or [] for c in rec.get('cells') or [] if isinstance(c, dict) and c.get('at')):
        return False, ' tidy_cell'
    return True, ''


def crop_harvest_ok(chain, cell, block):

def tidy_heads_at(chain, t_ms, skew_ms=60_000):
    """cropfarm-01: the tidy generations that may have been the HEAD when a dig happened at t_ms -> [gen]. Pure. The last one sited
    at or before t_ms (no `at`: in force) and every one sited within skew_ms after it (a clock skew: the head the bot read may be
    either -- Codex G9 r2 BLOCKER: the last in-force generation alone could be a FUTURE head). Their cells are judged together."""
    inf = sink_in_force(chain, t_ms, skew_ms)
    idx = max((i for i, g in enumerate(inf) if t_ms is None or not isinstance(g.get('at'), (int, float)) or g['at'] <= t_ms), default=None)
    return inf[idx:] if idx is not None else inf   # none sited before the dig: every one in the skew (Codex G9 r3)


def seedgrass_ok(chain, cell, block, t_ms, guard, park_box, tidy_chain=()):
    """cropfarm-01: is a `project=cropfarm op=clear kind=seedgrass` (the seed gather) a grass plant the gather may take? -> (bool, why).
    Pure. A SEED_GRASS block; never guarded (any project's record cell -- the farm's own too -- or a marker, when it was dug:
    tidy_guarded); never inside the park's box + 1; never a cell of the tidy generation in force when it was dug (`tidy_chain`: its
    fills -- the carried `prev` ones too --, cuts and clears, as the bot's gather guard reads the tidy head; tidy_guard leaves the
    tidy out -- Codex G9 r1 BLOCKER); within SEEDGRASS_R (Chebyshev) of the
    box of some generation of the farm's record. The protected zones are G3's."""
    if block not in SEED_GRASS:
        return False, ' not_grass'
    if tidy_guarded(cell, t_ms, guard):   # a record cell (the farm's own too) or a marker when it was dug
        return (False, ' guarded')
    if isinstance(park_box, list) and len(park_box) == 4 and park_box[0] - 1 <= cell[0] <= park_box[2] + 1 and park_box[1] - 1 <= cell[2] <= park_box[3] + 1:
        return (False, ' in_park')
    k = ck(cell)

def crew_overlaps(leases, chain, since_ms=None, helpers_max=CREW_HELPERS_MAX, gap=CREW_SEG_GAP, yield_ms=CREW_YIELD_MS, now_ms=None):
    """gkhelpers-01: THE GROUNDS CREW's lease rules over the tidy's leases -> [(name, gen, why)]. Pure.
    leases { name: [(gen, rec, released_ms)] } (read_projects); chain the tidy record's generations (a segment's columns are those of
    the generation in force when its lease was taken). Only leases taken at or after since_ms (the read's cut).
      - one holder on two crew leases at once (keeper and a helperN, or two helpers), each held [at, released or until]
      - a helperN with N > helpers_max
      - two holders on segments within `gap` columns (Chebyshev x, z) of each other, held together longer than yield_ms
    A lease still held is held up to NOW (now_ms, default the clock), never its future `until` (Codex r1 BLOCKER 2: a raced take
    read in the 50 ms before its yield measured 240 s of overlap)."""
    now_ms = int(time.time() * 1000) if now_ms is None else now_ms
    bad = []
    def spans(gens):
        """A lease's generations -> [(gen, rec, start, end)]: each held from its `at` to its release, else its `until` -- and never past
        the NEXT generation's `at` (a renewal supersedes it: only the newest generation is ever marked released; Claude r1 BLOCKERS
        1-2). A torn generation (no rec) is skipped."""
        gs = sorted(gens, key=lambda x: x[0])
        out_ = []
        for i, (g, r, rel) in enumerate(gs):
            if not r:
                continue
            end = int(rel) if rel is not None else int(r.get('until') or 0)
            nxt = next((int(r2.get('at') or 0) for _, r2, _ in gs[i + 1:] if r2), None)
            if nxt is not None:
                end = min(end, nxt)
            out_.append((g, r, int(r.get('at') or 0), min(end, now_ms)))
        return out_
    crew = []
    for name, gens in leases.items():
        m = re.match(r'^helper(\d+)$', name)
        if m and int(m.group(1)) > helpers_max:
            bad.append((name, None, 'helper beyond the crew cap %d' % helpers_max))
        if name == 'keeper' or m:
            for g, r, a, b in spans(gens):
                if since_ms is None or a >= since_ms:
                    crew.append((name, g, r.get('holder'), a, b))
    for i, (n0, g0, h0, a0, b0) in enumerate(crew):
        for n1, g1, h1, a1, b1 in crew[i + 1:]:
            if h0 == h1 and n0 != n1 and a0 < b1 and a1 < b0:
                bad.append((n1, g1, '%s held %s and %s at once' % (h1, n0, n1)))
    def cols_of(seg, at):
        rec = None
        for g in chain or []:
            if not isinstance(g.get('at'), (int, float)) or g['at'] <= at + 60_000:
                rec = g
        rec = rec or (chain[-1] if chain else None)
        return {(c['at'][0], c['at'][2]) for c in (rec or {}).get('cells') or [] if isinstance(c, dict) and c.get('seg') == seg and not c.get('prev') and c.get('at')}
    segs = []
    for name, gens in leases.items():
        if not name.startswith('seg-') or name == 'seg-clear':
            continue
        for g, r, a, b in spans(gens):
            if since_ms is None or a >= since_ms:
                segs.append((name, g, r.get('holder'), a, b, cols_of(name[4:], a)))
    for i, (n0, g0, h0, a0, b0, c0) in enumerate(segs):
        for n1, g1, h1, a1, b1, c1 in segs[i + 1:]:
            if h0 == h1 or min(b0, b1) - max(a0, a1) <= yield_ms:
                continue
            if any(max(abs(x - u), abs(z - v)) <= gap for x, z in c0 for u, v in c1):
                bad.append((n1, g1, '%s on %s and %s on %s within %d columns for %d ms' % (h0, n0, h1, n1, gap, min(b0, b1) - max(a0, a1))))
    return bad

YARD = 'yard'
YARD_H = 4
YARD_BAYS = {'yard@1': [((2, 3), (2, 12)), ((6, 7), (2, 12)), ((10, 11), (2, 12)), ((14, 15), (2, 12))],
             'yard@2': [((2, 3), (2, 23)), ((6, 7), (2, 23)), ((10, 11), (2, 23)), ((14, 15), (2, 23))]}

def yard_rot(orient, u, v):
    """townplan.mjs rot: (u, v) -> (dx, dz) for an orientation. Pure."""
    return {'E': (u, v), 'S': (-v, u), 'W': (-u, -v), 'N': (v, -u)}[orient]


def yard_slots(rec):
    """The yard record's pile slots (yard.mjs pileSlots) -> set of cell keys; empty when the record cannot say. Pure."""
    bays = YARD_BAYS.get(str(rec.get('blueprint') or ''))
    a, s0, o = rec.get('anchor'), rec.get('s'), rec.get('orient')
    if not bays or not isinstance(a, list) or len(a) < 2 or not isinstance(s0, int) or o not in ('E', 'S', 'W', 'N'):
        return set()
    out = set()
    for (u0, u1), (v0, v1) in bays:
        for k in range(1, YARD_H + 1):
            for v in range(v1, v0 + (k - 1) - 1, -1):
                for u in range(u0, u1 + 1):
                    dx, dz = yard_rot(o, u, v)
                    out.add(ck([a[0] + dx, s0 + k, a[1] + dz]))
    return out

def record_head(files):
    """files: [(gen, bytes)] -> the head record (buildproj.mjs recordHead, exactly: the root is the lowest gen with prev null;
    each later gen is the head only when its prev is the head and its prevSha the head's bytes' sha256) | None."""
    chain = record_chain(files)
    return chain[-1] if chain else None


def record_chain(files):
    """files: [(gen, bytes)] -> every generation of the valid chain, root first, each dict(rec, _gen, _sha) (record_head's walk:
    a generation joins only when its prev is the chain's last and its prevSha that one's bytes' sha256). Pure."""
    head, chain = None, []
    for gen, buf in sorted(files):
        try:
            rec = json.loads(buf.decode('utf-8'))
        except Exception:
            continue
        if not isinstance(rec, dict) or rec.get('gen') != gen:
            continue
        sha = hashlib.sha256(buf).hexdigest()
        if head is None:
            if rec.get('prev') is None:
                head = (gen, sha, rec)
                chain.append(dict(rec, _gen=gen, _sha=sha))
            continue
        if rec.get('prev') == head[0] and rec.get('prevSha') == head[1]:
            head = (gen, sha, rec)
            chain.append(dict(rec, _gen=gen, _sha=sha))
    return chain

BRIDGE_TORCH_EVERY = 8                 # bots/src/bridge.mjs TORCH_EVERY
BRIDGE_MERGE_R = 8                     # bots/src/bridge.mjs SITE_MERGE_R


def _cell(q):
    return isinstance(q, list) and len(q) == 3 and all(isinstance(v, int) for v in q)


def bridge_cells(span):
    """A bridge site's span (bots/src/bridge.mjs bridgeSpan, as the index stores it) -> [[x, y, z], ...]: EXACTLY the blocks its plan
    (bridgePlan) places, the bot's own dig guard (bridgeCellKeys) -- each row's centre deck, both side decks, both rails, and on every
    BRIDGE_TORCH_EVERY-th row (i % 8 == 0) the torch above the side +1 rail. Pure; a malformed span gives []."""
    out = []
    rows = span.get('rows') if isinstance(span, dict) else None
    for r in rows if isinstance(rows, list) else []:
        if not isinstance(r, dict):
            continue
        c = r.get('centre')
        if _cell(c):
            out.append(c)
        for sd in r.get('sides') if isinstance(r.get('sides'), list) else []:
            if not isinstance(sd, dict):
                continue
            for k in ('deck', 'rail'):
                if _cell(sd.get(k)):
                    out.append(sd[k])
            if sd.get('side') == 1 and _cell(sd.get('rail')) and isinstance(r.get('i'), int) and r['i'] % BRIDGE_TORCH_EVERY == 0:
                q = sd['rail']
                out.append([q[0], q[1] + 1, q[2]])
    return out


def read_bridges(d):
    """A town dir -> {cell key: ms first sighted} for every block of every bridge site ever sighted there (expedition-*.bridges.jsonl
    `sight` lines; bridgepost-01). Built or not, complete or broken, a site's cells are never the tidy's to dig (the bot's own guard,
    skills.mjs bridgeCellKeys: fail closed). An unparseable line is skipped. Never raises."""
    out = {}
    try:
        names = sorted(f for f in os.listdir(d) if BRIDGES_RE.match(f))
    except OSError:
        return out
    # the bot's sites (bridge.mjs bridgeSites): a sighting on the same axis whose near end is within BRIDGE_MERGE_R of a site's is that
    # site, and the site keeps its FIRST span -- only those cells are planned and guarded
    sites = []
    for f in names:
        try:
            lines = open(os.path.join(d, f), encoding='utf-8', errors='replace').read().split('\n')
        except OSError:
            continue
        for ln in lines:
            try:
                m = json.loads(ln) if ln.strip() else None
            except ValueError:
                continue
            if not isinstance(m, dict) or m.get('op') != 'sight':
                continue
            sp = m.get('span') if isinstance(m.get('span'), dict) else None
            if not sp or not _cell(sp.get('near')) or not isinstance(sp.get('axis'), list):
                continue
            t = m.get('t') if isinstance(m.get('t'), (int, float)) else 0
            same = [q for q in sites if q['axis'] == sp['axis'] and ((q['near'][0] - sp['near'][0]) ** 2 + (q['near'][2] - sp['near'][2]) ** 2) ** 0.5 <= BRIDGE_MERGE_R]

                continue
            sites.append({'axis': sp['axis'], 'near': sp['near'], 'span': sp, 't': t})
    for q in sites:
        for c in bridge_cells(q['span']):
            k = ck(c)
            out[k] = min(out.get(k, q['t']), q['t'])
    return out

def tidy_guard(pr):

    # earliest complete generation's completeAt), never from the retirement itself; with no complete demolition it stays reserved
    dj = (((pr or {}).get('projects') or {}).get(WALLDEMO) or {})

    done_at = min([g['completeAt'] for g in dchain if g.get('structure_complete') and isinstance(g.get('completeAt'), (int, float))] or [None], key=lambda v: float('inf') if v is None else v)
    for plot, pj in ((pr or {}).get('projects') or {}).items():

            # line, demolished) is no longer reserved once the demolition is complete: its ground is the groundskeeper's then (key
            # 'u|<cell>': reserved until)

                for c in rec.get('retired') or []:
                    if isinstance(c, dict) and c.get('at'):
                        u = 'u|' + ck(c['at'])
                        out[u] = min(out.get(u, ra), ra)
    return out


def zones_valid(raw):
    """THE BOT'S OWN SCHEMA (protectzone.mjs parseZones; Codex read r2 P2-3): an object, version 1, a `zones` list of at most 64,
    every zone an id string and a 6-integer ordered box within the spans, floor_y (if any) an integer inside it. Anything less
    makes the bot keep its last good list -- which this read cannot see -- so G3 cannot be judged: False (fail closed)."""
    isint = lambda v: isinstance(v, int) and not isinstance(v, bool)
    if not isinstance(raw, dict) or raw.get('version') != 1 or not isinstance(raw.get('zones'), list) or len(raw['zones']) > 64:
        return False
    for z in raw['zones']:
        b = z.get('box') if isinstance(z, dict) else None
        if not (isinstance(z, dict) and isinstance(z.get('id'), str) and z['id'] and isinstance(b, list) and len(b) == 6 and all(isint(v) for v in b)):
            return False
        x0, y0, z0, x1, y1, z1 = b
        if x0 > x1 or y0 > y1 or z0 > z1 or x1 - x0 > 1024 or z1 - z0 > 1024 or y1 - y0 > 448:
            return False
        fy = z.get('floor_y')
        if fy is not None and not (isint(fy) and y0 <= fy <= y1):
            return False
    return True


def lease_overlaps(leases, builders_max=BUILDERS_MAX):
    """{ name: [(gen, rec, released_ms)] } -> [(name, gen, why)]: a generation taken by ANOTHER holder while the one before it
    still blocked (unreleased and inside until + RECLAIM_MS), and any slot name beyond `builders_max` (the record's own: the
    ziggurat admits three)."""
    bad = []
    for name, gens in leases.items():
        m = re.match(r'^slot(\d+)$', name)
        if m and int(m.group(1)) > builders_max:
            bad.append((name, None, 'slot beyond builders_max'))
        gs = sorted(gens, key=lambda g: g[0])
        for (g0, r0, rel0), (g1, r1, _) in zip(gs, gs[1:]):
            if not r0 or not r1 or r0.get('holder') == r1.get('holder'):
                continue
            blocks_until = rel0 if rel0 is not None else (int(r0.get('until') or 0) + RECLAIM_MS)
            if int(r1.get('at') or 0) < blocks_until:
                bad.append((name, g1, '%s took it at %s, %s held it until %s' % (r1.get('holder'), r1.get('at'), r0.get('holder'), blocks_until)))
    return bad


def structcount_gens(root=os.path.expanduser('~/structcount/in')):
    """-> [(time, {town: day-doc})] oldest first, from the mirror's generation directories (gen-<YYYYmmddTHHMMSS.mmmZ>-<n>)."""
    out = []
    for d in glob.glob(os.path.join(root, 'gen-*')):
        m = re.match(r'^gen-(\d{8}T\d{6})', os.path.basename(d))
        if not m:
            continue
        try:
            doc = json.load(open(os.path.join(d, 'structcount-day.json')))
        except Exception:
            continue
        out.append((dt.datetime.strptime(m.group(1), '%Y%m%dT%H%M%S').replace(tzinfo=dt.timezone.utc), doc))
    return sorted(out, key=lambda g: g[0])


def load_surveys(root=SURVEY_DIR):
    """-> (present, {pool: [survey]}) from the conformance surveys' mirror (shopconform.py's JSON, one file each)."""
    if not os.path.isdir(root):
        return False, {}
    out = defaultdict(list)
    for f in glob.glob(os.path.join(root, '*.json')):
        try:
            d = json.load(open(f))
        except Exception:
            continue
        if isinstance(d, dict) and d.get('pool'):
            out[d['pool']].append(d)
    return True, dict(out)


def conformance(surveys, rec, since):
    """A completed town's CONFORMANCE from its surveys -> (ok, why). Pure. Counted: valid surveys of THIS record (gen and sha) whose
    observed time (`at` less SAVE_LAG_MIN) is after `since` (the completion). ok when two of them lie >= CONFORM_GAP_MIN apart with F1 >= CONFORM_F1
    each AND the newest counted survey has F1 >= CONFORM_F1 (damage after a good pair revokes it)."""
    if rec is None:
        return False, 'no record'
    # THE OBSERVED TIME is the survey's own time less SAVE_LAG_MIN: the copy holds the world as the server last SAVED it (a loaded,
    # changed chunk at its autosave, ~5 min; an unchanged or unloaded one exactly) -- never the region file's mtime, which a
    # quiet chunk never advances (a workshop that persists unchanged must still read as two observations an hour apart)
    def t(sv):
        # a time with no zone, or no time, is no observation (r6, Codex P2: a naive `at` raised TypeError)
        try:
            v = dt.datetime.fromisoformat(str(sv.get('at')).replace('Z', '+00:00'))
        except Exception:
            return None
        return v - dt.timedelta(minutes=SAVE_LAG_MIN) if v.tzinfo is not None else None

    def f1(sv):
        # a score that is not a finite number in [0, 1] is no score (r6, Codex P2: f1 "oops" raised ValueError)
        try:
            v = float(sv.get('f1'))
        except Exception:
            return None
        return v if v == v and 0.0 <= v <= 1.0 else None
    mine = sorted([sv for sv in surveys or [] if isinstance(sv, dict) and sv.get('valid') is True and sv.get('record_gen') == rec.get('_gen')
                   and sv.get('record_sha') == rec.get('_sha') and t(sv) is not None and t(sv) >= since and f1(sv) is not None], key=t)
    if not mine:
        return False, 'no valid survey of this record after the completion (%d surveys)' % len(surveys or [])
    newest = mine[-1]
    # the persistent pair lies AFTER the last bad survey (r6, Claude P3-4: good, damaged, repaired is not an hour of persistence);
    # a bad NEWEST survey leaves nothing after it (damage after a good pair revokes it)
    last_bad = max([i for i, sv in enumerate(mine) if f1(sv) < CONFORM_F1], default=-1)
    if last_bad == len(mine) - 1:
        return False, 'newest survey F1 %s at %s' % (newest.get('f1'), newest.get('at'))
    good = [sv for sv in mine[last_bad + 1:] if f1(sv) >= CONFORM_F1]
    if not good or (t(good[-1]) - t(good[0])).total_seconds() < CONFORM_GAP_MIN * 60:
        return False, '%d good survey(s), not %d min apart' % (len(good), CONFORM_GAP_MIN)
    return True, 'F1 %s .. %s over %d surveys, %s -> %s' % (good[0].get('f1'), newest.get('f1'), len(mine), good[0].get('at'), newest.get('at'))


def rooms_in_box(doc, box):
    """The structure count's lit rooms (B5 buildings: `building_rooms`, JSON strings with a `centre`) whose centre lies inside the
    record's cells' extent box = [x0, y0, z0, x1, y1, z1] (+-1) -> n, or None when the town's room data is missing or malformed
    (Codex read r3 P2-1: unknown is never zero). Record-SPECIFIC, height included (Codex read r3 P2-2: a room above or below
    does not count). Positive control, 10-25 on sandbox8 after SHOP-E2E: one lit room, centre 734.4,121,764, inside the record's
    extent; the base world: none."""
    rooms = (doc or {}).get('building_rooms') if isinstance(doc, dict) else None
    if not isinstance(rooms, list) or not box or len(box) != 6:
        return None
    n = 0
    for raw in rooms:
        try:
            r = json.loads(raw) if isinstance(raw, str) else raw
            cx, cy, cz = (float(v) for v in r.get('centre'))
        except Exception:
            return None
        if r.get('lit') and box[0] - 1 <= cx <= box[3] + 1 and box[1] - 1 <= cy <= box[4] + 1 and box[2] - 1 <= cz <= box[5] + 1:
            n += 1
    return n


def record_extent(rec):
    """The record's cells' extent -> [x0, y0, z0, x1, y1, z1] | None."""
    cells = [c.get('at') for c in (rec or {}).get('cells') or [] if isinstance(c, dict) and isinstance(c.get('at'), list) and len(c['at']) == 3]
    if not cells:
        return None
    return [min(c[0] for c in cells), min(c[1] for c in cells), min(c[2] for c in cells), max(c[0] for c in cells), max(c[1] for c in cells), max(c[2] for c in cells)]


def world_evidence(gens, cut, complete_at, boxes=None):
    """gens: structcount_gens(); cut: the deploy time; complete_at: {pool: first completion time}; boxes: {pool: the record's box}.
    -> ({pool: why}, litter). A completed town is CONFIRMED when the first count after its completion shows a lit room INSIDE ITS
    RECORD'S BOX and the last count before the deploy showed none there. litter: {pool: (before, after)} for every town both
    counts cover (the no-harm line's cells)."""
    boxes = boxes or {}
    before = [g for g in gens if g[0] < cut]
    base = before[-1][1] if before else None
    world = {}
    for pool, t in complete_at.items():
        after = [g for g in gens if g[0] > t]
        if base is None:
            world[pool] = 'no count before the deploy'
        elif not after:
            world[pool] = 'no count after the completion yet'
        elif not boxes.get(pool):
            world[pool] = 'no record box'
        else:
            r0, r1 = rooms_in_box(base.get(pool), boxes[pool]), rooms_in_box(after[0][1].get(pool), boxes[pool])
            if r0 is None or r1 is None:
                world[pool] = 'room data missing or malformed (before %s, after %s): unknown' % (r0, r1)
            else:
                world[pool] = 'room' if r0 == 0 and r1 >= 1 else 'lit rooms in the record extent %d -> %d' % (r0, r1)
    litter = {}
    last = gens[-1][1] if gens and gens[-1][0] > cut else None
    if base is not None and last is not None:
        for pool in set(base) & set(last):
            a, b = (base.get(pool) or {}).get('litter_cells'), (last.get(pool) or {}).get('litter_cells')
            if isinstance(a, int) and isinstance(b, int):
                litter[pool] = (a, b)
    return world, litter


def pos_of(r):
    p = r.get('pos')
    if isinstance(p, dict) and all(k in p for k in 'xyz'):
        return [int(p['x'] // 1), int((p['y'] + 1e-6) // 1), int(p['z'] // 1)]
    if isinstance(p, (list, tuple)) and len(p) == 3:

    return None

def combine(out, lists):
    g = lambda k: out.get(k, 0)
    res = {k: v for k, v in out.items()}
    for k in ('off_record', 'clear_outside', 'zone', 'lease_overlap', 'felled_unplanted', 'g5_unjudged', 'unknown_stood', 'bag_loss',
              'bag_unjudged', 'oracle_missing', 'zone_unknown', 'placed_twice', 'kept_thrown', 'kept_unjudged'):
        res[k] = g(k)
    res['arms_breach'] = g('arms_rows_control_post')
    res['liveness'] = int(g('licence_canary') >= 1 and g('visits_canary') >= 1)
    # EXPOSURE IS RECORD CONFORMANCE AND PERSISTENCE (Codex read r3; owner 10-25): two surveys >= 1 h apart with F1 >= 0.98 and the
    # newest still >= 0.98, per town; no survey mirror -> never ready (fail closed). The room check stays PRELIMINARY, reported.
    res['conformance_source'] = g('conformance_source')
    res['exposure_ready'] = int(res['conformance_source'] == 1 and g('complete_towns_conform') >= EXPOSE_TOWNS)
    res['instrument_control'] = int(g('craft_post_control') >= 1 and g('gather_post_control') >= 1)
    for name, key in (('gather_ok', 'gather'), ('rung_ok', 'rung')):
        cells = [(g('%s_ok_pre_canary' % key), g('%s_pre_canary' % key)), (g('%s_ok_post_canary' % key), g('%s_post_canary' % key)),
                 (g('%s_ok_pre_control' % key), g('%s_pre_control' % key)), (g('%s_ok_post_control' % key), g('%s_post_control' % key))]
        d, z = did_z(cells)
        res['%s_did' % name] = d; res['%s_z' % name] = z
        res[name] = 1 if d is None or z is None or not (d < TOL and z < -2) else 0
    lit = lists.get('_litter') or {}
    can = [b - a for p, (a, b) in lit.items() if p in set(lists.get('_cans') or [])]
    con = [b - a for p, (a, b) in lit.items() if p not in set(lists.get('_cans') or [])]
    res['litter_did'] = (sum(can) / len(can) - sum(con) / len(con)) if can and con else None
    res['_lists'] = dict(lists)
    return res


def verdict_of(res, name):
    if res.get('%s_did' % name) is None or res.get('%s_z' % name) is None:
        return 'UNEXPOSED'
    return 'ok' if res[name] else 'DOWN'

        CAN = man['canary_pool']; CV = man.get('canary_code_version') or ''
    CANS = {x.strip() for x in str(CAN).split(',') if x.strip()}
    now = dt.datetime.now(dt.timezone.utc)
    elapsed = (now - CUT).total_seconds() / 60
    assert elapsed > 0 and CAN, 'no canary declared'
    W = min(elapsed, float(sys.argv[1]) if len(sys.argv) > 1 and not sys.argv[1].startswith('-') else 360)
    END, PRE = CUT + dt.timedelta(minutes=W), CUT - dt.timedelta(minutes=W)

    def load_window(since, until):
        ev = Events.load(paths='/var/log/mcai/*/skill-*.jsonl', since=since, until=until)
        key = lambda r: (str(r.get('t')), ((r.get('bot') or {}).get('name')), r.get('name'), r.get('detail'))
        out, seen = [], set()
        for r in ev.rows:
            if key(r) not in seen:
                out.append(r); seen.add(key(r))
        for k in range(0, (until.date() - since.date()).days + 1):
            tag = (since.date() + dt.timedelta(days=k + 1)).strftime('%Y%m%d')
            for g in glob.glob('/var/log/mcai/*/skill-*.jsonl-%s.gz' % tag):
                for r in Events.load(paths=g, since=since, until=until).rows:
                    if key(r) not in seen:
                        out.append(r); seen.add(key(r))
        return out

    raw = load_window(PRE, END)
    rows, walked, offbuild = [], 0, 0
    bots = defaultdict(set)
    for r in raw:
        b = (r.get('bot') or {}).get('name')
        t = r.get('t')
        if not b or t is None:
            continue
        walked += 1
        arm = 'canary' if pool_of(b) in CANS else 'control'
        per = 'post' if t >= CUT else 'pre'
        ver = (((r.get('raw') or {}).get('code') or {}).get('version') or '')
        if arm == 'canary' and per == 'post' and CV and ver and not ver.startswith(CV):
            offbuild += 1
            continue
        bots[(per, arm)].add(b)
        sk = ((r.get('raw') or {}).get('skill') or {})
        name = sk.get('name') or r.get('name')
        feet = pos_of({'pos': (r.get('bot') or {}).get('pos')}) if arm == 'canary' and per == 'post' else None
        if name in ROW_KINDS or feet:
            item = (sk.get('args') or {}).get('item') if isinstance(sk.get('args'), dict) else None
            rows.append({'t': t, 'per': per, 'arm': arm, 'name': name if name in ROW_KINDS else '_pos', 'status': sk.get('status'),
                         'detail': (r.get('detail') or '') if name in ROW_KINDS else '', 'bot': b, 'item': item, 'feet': feet})
    projects = read_projects()
    plans = read_plans()
    zones_by_pool = {}
    for pool, keys in plans.items():
        for key, info in keys.items():
            for wkey in info['worlds']:
                world = json.loads(wkey) if wkey != '"__unparseable__"' else None
                zones_by_pool.setdefault(pool, []).extend(zones_of(info['dir'], world))
    gens = structcount_gens()
    surveys = load_surveys()
    out, lists = judge(rows, projects, zones_by_pool, cut=CUT, gens=gens, surveys=surveys)
    res = combine(out, lists)
    L = res['_lists']
    hrs = {k: len(v) * W / 60 for k, v in bots.items()}
    recs = {p: pr['record'] for p, pr in projects.items() if pr.get('record')}
    G = lambda k: out.get(k, 0)
    print('DENOMINATORS rows walked %d (canary rows on another build skipped %d) | bots pre canary %d control %d | post canary %d control %d | window %d min | NOMINAL bot-h (bots x window) post canary %.1f control %.1f'
          % (walked, offbuild, len(bots[('pre', 'canary')]), len(bots[('pre', 'control')]), len(bots[('post', 'canary')]), len(bots[('post', 'control')]), W,
             hrs.get(('post', 'canary'), 0), hrs.get(('post', 'control'), 0)))
    print('             oracle: towns with a workshop record %d (canary %d) | canary towns mutating %d | lease files %d over %d names | canary build rows with a position %d | structure counts %d (newest %s)'
          % (len(recs), sum(1 for p in recs if p in CANS), G('towns_mutating'), G('lease_files'), G('lease_names'), G('build_rows_with_pos'), len(gens), gens[-1][0].strftime('%m-%d %H:%MZ') if gens else '-'))
    print('LIVENESS     licence rows canary %d | build visits canary %d -> %s' % (G('licence_canary'), G('visits_canary'), 'ok' if res['liveness'] else 'INERT'))
    print('VISITS       stops %s | placed %d refused %d silent %d cleared %d planted %d unknown %d | sited %d | bag rows %d (names %d, cobble/planks scaffold %d)'
          % ({k[6:]: v for k, v in sorted(out.items()) if k.startswith('visit_')}, G('sum_placed'), G('sum_refused'), G('sum_silent'),
             G('sum_cleared'), G('sum_planted'), G('sum_unknown'), G('sited'), G('bag_rows'), G('bag_names'), G('bag_scaffold')))
    print('             stops other than complete (first 6) %s' % L.get('why', [])[:6])
    print('ARMS         control build rows after the deploy %d (must be 0) %s' % (res['arms_breach'], L.get('control_rows', [])[:3]))
    print('ORACLE       towns mutating without a readable record %d %s | without a readable zones file %d %s'
          % (res['oracle_missing'], L.get('oracle_missing', [])[:3], res['zone_unknown'], L.get('zone_unknown', [])[:3]))
    print('DEFECTS      G1 off the record %d of %d placements %s | G2 clears outside the list %d of %d %s | G3 in a zone %d of %d mutations %s'
          % (res['off_record'], G('placements'), L.get('off_record', [])[:3], res['clear_outside'], G('clears'), L.get('clear_outside', [])[:3], res['zone'], G('mutations'), L.get('zone', [])[:3]))
    print('             G4 lease overlaps %d of %d files %s | G5 fellings with a replant cell unplanted %d of %d tree clears (unjudged %d) %s'
          % (res['lease_overlap'], G('lease_files'), L.get('lease_overlap', [])[:3], res['felled_unplanted'], G('tree_clears'), res['g5_unjudged'], L.get('felled_unplanted', [])[:3]))
    print('             G6 unknown cells stood on %d of %d unknown rows %s | G8 bag losses %d of %d names (%d of %d bag rows unjudged: not a clean exit %s) | REPORTED G7 placed twice %d %s'

          % tuple([x for n in ('gather_ok', 'rung_ok') for x in ('null' if res['%s_did' % n] is None else '%+.3f' % res['%s_did' % n], verdict_of(res, n))]
                  + ['null' if res['litter_did'] is None else '%+.1f' % res['litter_did']]))
    print('             (gather canary %d/%d -> %d/%d control %d/%d -> %d/%d | rungs canary %d/%d -> %d/%d control %d/%d -> %d/%d ok/rows; UNEXPOSED = a cell under %d rows, never ok)'
