#!/usr/bin/env python3
# shopread.py [window_min] [--selftest] -- the read for canary `shop-01` (the building tier's WORKSHOP: a fitted, frozen record
# per town, built by up to two builders on slot and segment leases; design docs/reports/building-tier-2026-10-23.md section 6.1
# row 3). CANARY_DRYRUN=pool[,pool]:sha:iso for dry runs (never emits). Judges the CANARY ARM; DiD against control for the
# no-harm lines; DENOMINATORS FIRST. Needs townplanread.py beside it (kv, pool_of, read_plans, zones_of, in_box, did_z, ck).
#
# THE CHANGE (what only this build writes):
#   _build_mode          once per spawn: the licence (phase=mode v=1 shop= blueprints= builders_max= reserve=); also marks a NEW
#                        CONNECTION of that bot (its unknown cells start empty)
#   _build               op=sited|place|replant|clear|unknown|known  project seg cell block kind saplings gen
#   _build_seg           one per segment worked in a visit: seg open done
#   _build_visit         every build_shop run, every exit: outcome stop placed refused silent cleared planted unknown slot gen took back
#   _build_bag           one per visit that reached its bag: name=bag0/took/crafted/placed/back/bag1 (shop.mjs bagBalance)
#   _structure_complete  the record's completion transition, once per town
# THE TOWN FILES (the oracle; /var/lib/mcai/_pool-<pool>/): project-<home>.shop.g<N>.json (the record chain: gen, prev, prevSha
#   sha256 of the bytes), project-<home>.shop.<slotK|seg-X>.lease.g<N>.json (+ .released), protected-zones.json.
# THE WORLD (the structure count's mirror on this host, ~/structcount/in/gen-<UTC>/structcount-day.json, one per night).
#
# GATES (deterministic; each has a failing fixture and a passing control in --selftest)
#   G1 off_record      a `_build op=place` cell that is not a cell of the town's record
#   G2 clear_outside   a `_build op=clear` cell outside the record's clear list (kind=foreign: outside the record's cells)
#   G3 zone            a placement, replant or clear inside a protected zone
#   G4 lease_overlap   two holders live on one lease (a later generation by ANOTHER holder taken before the earlier one was
#                      released or lapsed + RECLAIM_MS), or a slot name beyond builders_max
#   G5 felled_unplanted  a `kind=tree` clear whose row says fewer replant cells read a sapling than the record has (saplings=a/b,
#                      a < b); a tree clear without the field is g5_unjudged (also a gate)
#   G6 unknown_stood   a bot's OWN build row whose position (its stand) is IN or ON a cell its own `op=unknown` row named, before
#                      an `op=known`/clear of it or its next `_build_mode` (a new connection)
#   G8 bag_loss        a `_build_bag` part (any name but cobblestone and planks, which the pathfinder may scaffold) the visit
#                      cannot account for (unexplained < 0) -- judged only on a clean exit with silent=0 (bag_unjudged: reported)
#   ORACLE             oracle_missing: a canary town with build mutations and no readable record; zone_unknown: ... and a
#                      protected-zones.json that is present but unreadable (a MISSING file is no zones, as the bot reads it)
#   ARMS               control rows of these kinds after the deploy (must be 0; build_shop is chatOnly -- a player's chat to a
#                      CONTROL bot writes `_build_visit outcome=off`: read that row before calling it a leak)
#   LIVENESS           canary `_build_mode` >= 1 and canary `_build_visit` >= 1
#   EXPOSURE           WORLD EVIDENCE (read r1, both engines: a completion row is the code's own claim): a canary town counts
#                      when it has a `_structure_complete` row AND the first structure count after that row shows a LIT ROOM
#                      INSIDE ITS RECORD'S BOX (rooms_in_box; none there before the deploy); >= EXPOSE_TOWNS such towns. NOT MEASURED
#                      (named): the design's PERSISTENT rule and record conformance F1 >= 0.98 (a `structcount scan --record`
#                      is not built)
#   NO HARM (WATCH, DiD of success shares; fails only below its tolerance with z < -2; UNEXPOSED under MIN_CELL)
#     gather_ok, rung_ok (wooden/stone/iron pickaxe crafts); litter cells (structcount, canary vs control towns, reported);
#     deaths: the default gate (two-death floor); immobile: immobiledid
# REPORTED, NOT A GATE: G7 placed_twice (a cell placed twice within 10 min: physically a re-placement after the cell was emptied
#   -- a repair or a survival dig -- never two blocks in one cell; read r1 both engines).
import sys, os, json, re, glob, hashlib
import datetime as dt
from collections import Counter, defaultdict

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from townplanread import (kv, pool_of, read_plans, zones_of, in_box, did_z, ck, MIN_CELL)  # noqa: E402

TOL = -0.20
RUNGS = ('wooden_pickaxe', 'stone_pickaxe', 'iron_pickaxe')
BUILDERS_MAX = 2
RECLAIM_MS = 60_000
PLACE_TWICE_MIN = 10
EXPOSE_TOWNS = 2
BUILD_KINDS = ('_build_mode', '_build', '_build_seg', '_build_visit', '_build_bag', '_structure_complete')
ROW_KINDS = BUILD_KINDS + ('craft', 'gather')
REC_RE = re.compile(r'^(project-(-?\d+)_(-?\d+)_(-?\d+)\.shop)\.g(\d+)\.json$')
LEASE_RE = re.compile(r'^(project-(-?\d+)_(-?\d+)_(-?\d+)\.shop)\.(slot\d+|seg-[A-Za-z0-9-]+)\.lease\.g(\d+)\.json$')   # wall-F: upper case (Claude read r1 P2-1)
SCAFFOLD_NAMES = {'cobblestone'} | {'%s_planks' % w for w in ('oak', 'spruce', 'birch', 'jungle', 'acacia', 'dark_oak', 'mangrove', 'cherry', 'pale_oak', 'bamboo', 'crimson', 'warped')}
BAG_RE = re.compile(r'(?:^| )([a-z_]+)=(-?\d+)/(-?\d+)/(-?\d+)/(-?\d+)/(-?\d+)/(-?\d+)')


def xyz_of(s):
    try:
        a = [int(v) for v in str(s).split(',')]
        return a if len(a) == 3 else None
    except ValueError:
        return None


def record_head(files):
    """files: [(gen, bytes)] -> the head record (buildproj.mjs recordHead, exactly: the root is the lowest gen with prev null;
    each later gen is the head only when its prev is the head and its prevSha the head's bytes' sha256) | None."""
    head = None
    for gen, buf in sorted(files):
        try:
            rec = json.loads(buf.decode('utf-8'))
        except Exception:
            continue
        if not isinstance(rec, dict) or rec.get('gen') != gen:
            continue
        sha = hashlib.sha256(buf).hexdigest()
        if head is None:
            if rec.get('prev') is None:
                head = (gen, sha, rec)
            continue
        if rec.get('prev') == head[0] and rec.get('prevSha') == head[1]:
            head = (gen, sha, rec)
    return head[2] if head else None


def read_projects(root='/var/lib/mcai'):
    """-> { pool: { 'record': rec | None, 'leases': { name: [(gen, rec, released_ms|None)] }, 'dir': d } } for every town."""
    out = {}
    for d in glob.glob(os.path.join(root, '_pool-*')):
        name = os.path.basename(d)[len('_pool-'):]
        pool = pool_of(name)
        try:
            names = os.listdir(d)
        except OSError:
            continue
        recs, leases = [], defaultdict(list)
        for f in names:
            m = REC_RE.match(f)
            if m:
                try:
                    recs.append((int(m.group(5)), open(os.path.join(d, f), 'rb').read()))
                except OSError:
                    pass
                continue
            m = LEASE_RE.match(f)
            if m:
                p = os.path.join(d, f)
                try:
                    rec = json.load(open(p))
                except Exception:
                    rec = None
                rel = None
                try:
                    rel = int(open(p + '.released').read().strip() or 0) or int(os.path.getmtime(p + '.released') * 1000)
                except (OSError, ValueError):
                    rel = None
                leases[m.group(5)].append((int(m.group(6)), rec, rel))
        # the bot's own rule (skills.mjs townPlanZones): a MISSING file is known and empty; a present file that does not parse
        # to an object is unreadable -- the bot then uses its last good list or refuses, and this read cannot judge G3
        zf = os.path.join(d, 'protected-zones.json')
        if not os.path.exists(zf):
            zok = True
        else:
            try:
                zok = zones_valid(json.load(open(zf)))
            except Exception:
                zok = False
        out[pool] = {'record': record_head(recs) if recs else None, 'records_files': len(recs), 'leases': dict(leases), 'dir': d, 'zones_ok': zok}
    return out


def zones_valid(raw):
    """THE BOT'S OWN SCHEMA (protectzone.mjs parseZones; Codex read r2 P2-3): an object, version 1, a `zones` list of at most 64,
    every zone an id string and a 6-integer ordered box within the spans, floor_y (if any) an integer inside it. Anything less
    makes the bot keep its last good list -- which this read cannot see -- so G3 cannot be judged: False (fail closed)."""
    isint = lambda v: isinstance(v, int) and not isinstance(v, bool)
    if not isinstance(raw, dict) or raw.get('version') != 1 or not isinstance(raw.get('zones'), list) or len(raw['zones']) > 64:
        return False
    for z in raw['zones']:
        b = z.get('box') if isinstance(z, dict) else None
        if not (isinstance(z, dict) and isinstance(z.get('id'), str) and z['id'] and isinstance(b, list) and len(b) == 6 and all(isint(v) for v in b)):
            return False
        x0, y0, z0, x1, y1, z1 = b
        if x0 > x1 or y0 > y1 or z0 > z1 or x1 - x0 > 1024 or z1 - z0 > 1024 or y1 - y0 > 448:
            return False
        fy = z.get('floor_y')
        if fy is not None and not (isint(fy) and y0 <= fy <= y1):
            return False
    return True


def lease_overlaps(leases):
    """{ name: [(gen, rec, released_ms)] } -> [(name, gen, why)]: a generation taken by ANOTHER holder while the one before it
    still blocked (unreleased and inside until + RECLAIM_MS), and any slot name beyond BUILDERS_MAX."""
    bad = []
    for name, gens in leases.items():
        m = re.match(r'^slot(\d+)$', name)
        if m and int(m.group(1)) > BUILDERS_MAX:
            bad.append((name, None, 'slot beyond builders_max'))
        gs = sorted(gens, key=lambda g: g[0])
        for (g0, r0, rel0), (g1, r1, _) in zip(gs, gs[1:]):
            if not r0 or not r1 or r0.get('holder') == r1.get('holder'):
                continue
            blocks_until = rel0 if rel0 is not None else (int(r0.get('until') or 0) + RECLAIM_MS)
            if int(r1.get('at') or 0) < blocks_until:
                bad.append((name, g1, '%s took it at %s, %s held it until %s' % (r1.get('holder'), r1.get('at'), r0.get('holder'), blocks_until)))
    return bad


def structcount_gens(root=os.path.expanduser('~/structcount/in')):
    """-> [(time, {town: day-doc})] oldest first, from the mirror's generation directories (gen-<YYYYmmddTHHMMSS.mmmZ>-<n>)."""
    out = []
    for d in glob.glob(os.path.join(root, 'gen-*')):
        m = re.match(r'^gen-(\d{8}T\d{6})', os.path.basename(d))
        if not m:
            continue
        try:
            doc = json.load(open(os.path.join(d, 'structcount-day.json')))
        except Exception:
            continue
        out.append((dt.datetime.strptime(m.group(1), '%Y%m%dT%H%M%S').replace(tzinfo=dt.timezone.utc), doc))
    return sorted(out, key=lambda g: g[0])


def rooms_in_box(doc, box):
    """The structure count's lit, roofed rooms (B5 buildings: `building_rooms`, JSON strings with a `centre`) whose centre lies in
    the record's box [x0, z0, x1, z1] (+-1) -> n. Record-SPECIFIC world evidence (Codex read r2 P2-5): another town building
    rising elsewhere does not count. Positive control, 10-25 on sandbox8 after SHOP-E2E: one lit room, centre 734.4,121,764 in the
    record box [731,760,737,768]; the base world: none."""
    n = 0
    for raw in (doc or {}).get('building_rooms') or []:
        try:
            r = json.loads(raw) if isinstance(raw, str) else raw
            cx, _, cz = r.get('centre')
        except Exception:
            continue
        if r.get('lit') and box and box[0] - 1 <= cx <= box[2] + 1 and box[1] - 1 <= cz <= box[3] + 1:
            n += 1
    return n


def world_evidence(gens, cut, complete_at, boxes=None):
    """gens: structcount_gens(); cut: the deploy time; complete_at: {pool: first completion time}; boxes: {pool: the record's box}.
    -> ({pool: why}, litter). A completed town is CONFIRMED when the first count after its completion shows a lit room INSIDE ITS
    RECORD'S BOX and the last count before the deploy showed none there. litter: {pool: (before, after)} for every town both
    counts cover (the no-harm line's cells)."""
    boxes = boxes or {}
    before = [g for g in gens if g[0] < cut]
    base = before[-1][1] if before else None
    world = {}
    for pool, t in complete_at.items():
        after = [g for g in gens if g[0] > t]
        if base is None:
            world[pool] = 'no count before the deploy'
        elif not after:
            world[pool] = 'no count after the completion yet'
        elif not boxes.get(pool):
            world[pool] = 'no record box'
        else:
            r0, r1 = rooms_in_box(base.get(pool), boxes[pool]), rooms_in_box(after[0][1].get(pool), boxes[pool])
            world[pool] = 'confirmed' if r0 == 0 and r1 >= 1 else 'lit rooms in the record box %d -> %d' % (r0, r1)
    litter = {}
    last = gens[-1][1] if gens and gens[-1][0] > cut else None
    if base is not None and last is not None:
        for pool in set(base) & set(last):
            a, b = (base.get(pool) or {}).get('litter_cells'), (last.get(pool) or {}).get('litter_cells')
            if isinstance(a, int) and isinstance(b, int):
                litter[pool] = (a, b)
    return world, litter


def pos_of(r):
    p = r.get('pos')
    if isinstance(p, dict) and all(k in p for k in 'xyz'):
        return [int(p['x'] // 1), int((p['y'] + 1e-6) // 1), int(p['z'] // 1)]
    if isinstance(p, (list, tuple)) and len(p) == 3:
        return [int(p[0] // 1), int((p[1] + 1e-6) // 1), int(p[2] // 1)]
    return None


def judge(rows, projects, zones_by_pool, cut=None, gens=()):
    """rows: [{t, per, arm, name, status, detail, bot, item, feet}]; projects: read_projects(); zones_by_pool: {pool: [box]};
    gens: structcount_gens() (the world evidence; empty: none)."""
    out = Counter()
    lists = defaultdict(list)
    rows = sorted(rows, key=lambda r: r['t'])
    placed = defaultdict(list)          # (pool, cell) -> [t]
    unknown_open = {}                   # (bot, cell) -> t
    complete_at = {}
    mutating, cans = set(), set()
    for r in rows:
        f = kv(r['detail'])
        arm, per, name, pool, bot = r['arm'], r['per'], r['name'], pool_of(r['bot']), r['bot']
        if name in BUILD_KINDS and arm == 'control' and per == 'post':
            out['arms_rows_control_post'] += 1
            if name == '_build_visit':
                lists['control_rows'].append('%s %s' % (bot, f.get('outcome')))
        if name == 'craft':
            out['craft_%s_%s' % (per, arm)] += 1
            if r.get('item') in RUNGS:
                out['rung_%s_%s' % (per, arm)] += 1
                out['rung_ok_%s_%s' % (per, arm)] += int(r['status'] == 'success')
        if name == 'gather':
            out['gather_%s_%s' % (per, arm)] += 1
            out['gather_ok_%s_%s' % (per, arm)] += int(r['status'] == 'success')
        if arm == 'canary':
            cans.add(pool)
        if arm != 'canary' or per != 'post':
            continue
        if name == '_build_mode':
            for k in [k for k in unknown_open if k[0] == bot]:      # a new connection: its unknown set starts empty -- FIRST
                del unknown_open[k]
        # G6 on the bot's OWN build rows only (their position is the builder's stand): feet IN the cell or ON it
        if name in BUILD_KINDS and r.get('feet') and unknown_open:
            p = r['feet']
            for cand in (ck(p), ck([p[0], p[1] - 1, p[2]])):
                if (bot, cand) in unknown_open and r['t'] > unknown_open[(bot, cand)]:
                    out['unknown_stood'] += 1; lists['unknown_stood'].append('%s at %s' % (bot, cand))
                    del unknown_open[(bot, cand)]
        if name in BUILD_KINDS and r.get('feet'):
            out['build_rows_with_pos'] += 1
        if name == '_build_mode':
            out['licence_canary'] += 1
        elif name == '_build_visit':
            out['visits_canary'] += 1
            out['visit_%s' % f.get('stop', '?')] += 1
            for k in ('placed', 'refused', 'silent', 'cleared', 'planted', 'unknown'):
                try:
                    out['sum_%s' % k] += int(f.get(k) or 0)
                except ValueError:
                    pass
            if f.get('stop') not in ('complete', None):
                lists['why'].append('%s %s:%s' % (bot, f.get('stop'), (f.get('why') or '-')[:50]))
        elif name == '_build_bag':
            out['bag_rows'] += 1
            # G8 judges only a CLEAN exit with no silent click (Claude read r2 P2: a late-acked silent click placed its block
            # untallied; a death or an abort leaves the bag mid-visit) -- the rest are reported, never a loss
            if f.get('exit') in (None, '-', 'aborted', 'error', 'died', 'off') or f.get('silent') != '0':
                out['bag_unjudged'] += 1; lists['bag_unjudged'].append('%s exit=%s silent=%s' % (bot, f.get('exit'), f.get('silent')))
                continue
            for m in BAG_RE.finditer(r['detail'] or ''):
                n = m.group(1)
                b0, took, crafted, pl, back, b1 = (int(m.group(i)) for i in range(2, 8))
                unexplained = b1 - (b0 + took + crafted - pl - back)
                out['bag_names'] += 1
                if unexplained < 0 and n not in SCAFFOLD_NAMES:
                    out['bag_loss'] += 1; lists['bag_loss'].append('%s %s %d' % (bot, n, unexplained))
                elif unexplained < 0:
                    out['bag_scaffold'] += -unexplained
        elif name == '_structure_complete':
            complete_at.setdefault(pool, r['t'])
        elif name == '_build':
            op, cell = f.get('op'), xyz_of(f.get('cell'))
            pr = projects.get(pool) or {}
            rec = pr.get('record')
            if op == 'sited':
                out['sited'] += 1
                continue
            if cell is None:
                continue
            k = ck(cell)
            if op in ('place', 'clear', 'replant'):
                mutating.add(pool)
                out['mutations'] += 1
                if any(in_box(b, cell) for b in zones_by_pool.get(pool) or []):
                    out['zone'] += 1; lists['zone'].append('%s %s %s' % (bot, op, k))
            if op == 'place':
                out['placements'] += 1
                if rec is not None and k not in {ck(c['at']) for c in rec.get('cells') or []}:
                    out['off_record'] += 1; lists['off_record'].append('%s %s %s' % (bot, k, f.get('block')))
                if [t for t in placed[(pool, k)] if (r['t'] - t).total_seconds() < PLACE_TWICE_MIN * 60]:
                    out['placed_twice'] += 1; lists['placed_twice'].append('%s %s' % (pool, k))
                placed[(pool, k)].append(r['t'])
            elif op == 'replant':
                out['replants'] += 1
            elif op == 'clear':
                out['clears'] += 1
                if rec is not None:
                    if f.get('kind') == 'foreign':
                        ok = k in {ck(c['at']) for c in rec.get('cells') or []}
                    else:
                        ok = k in {ck(c['at']) for c in rec.get('clear') or []}
                    if not ok:
                        out['clear_outside'] += 1; lists['clear_outside'].append('%s %s %s %s' % (bot, k, f.get('kind'), f.get('block')))
                if f.get('kind') == 'tree':
                    out['tree_clears'] += 1
                    m = re.match(r'^(\d+)/(\d+)$', f.get('saplings') or '')
                    need = len(rec.get('replant') or []) if rec is not None else None
                    if not m or int(m.group(1)) > int(m.group(2)) or (need is not None and int(m.group(2)) != need):
                        out['g5_unjudged'] += 1; lists['g5_unjudged'].append('%s %s saplings=%s record=%s' % (bot, k, f.get('saplings'), need))
                    elif int(m.group(1)) < int(m.group(2)):
                        out['felled_unplanted'] += 1; lists['felled_unplanted'].append('%s %s saplings=%s' % (bot, k, f.get('saplings')))
                unknown_open.pop((bot, k), None)
            elif op == 'unknown':
                out['unknown_rows'] += 1
                unknown_open[(bot, k)] = r['t']
            elif op == 'known':
                unknown_open.pop((bot, k), None)
    for pool in sorted(mutating):
        pr = projects.get(pool) or {}
        if pr.get('record') is None:
            out['oracle_missing'] += 1; lists['oracle_missing'].append('%s (record files %s)' % (pool, pr.get('records_files', 0)))
        if not pr.get('zones_ok'):
            out['zone_unknown'] += 1; lists['zone_unknown'].append(pool)
    out['towns_mutating'] = len(mutating)
    for pool, pr in projects.items():
        leases = pr.get('leases') or {}
        out['lease_files'] += sum(len(v) for v in leases.values())
        out['lease_names'] += len(leases)
        for name, gen, why in lease_overlaps(leases):
            out['lease_overlap'] += 1; lists['lease_overlap'].append('%s %s g%s: %s' % (pool, name, gen, why))
    out['complete_towns'] = len(complete_at)
    lists['complete_towns'] = sorted(complete_at)
    boxes = {p: (pr.get('record') or {}).get('box') for p, pr in projects.items()}
    world, litter = world_evidence(list(gens), cut, complete_at, boxes) if cut is not None else ({}, {})
    out['complete_towns_world'] = sum(1 for v in world.values() if v == 'confirmed')
    lists['world'] = ['%s: %s' % kv_ for kv_ in sorted(world.items())]
    lists['_litter'] = litter
    lists['_cans'] = sorted(cans)
    return out, lists


def combine(out, lists):
    g = lambda k: out.get(k, 0)
    res = {k: v for k, v in out.items()}
    for k in ('off_record', 'clear_outside', 'zone', 'lease_overlap', 'felled_unplanted', 'g5_unjudged', 'unknown_stood', 'bag_loss',
              'bag_unjudged', 'oracle_missing', 'zone_unknown', 'placed_twice'):
        res[k] = g(k)
    res['arms_breach'] = g('arms_rows_control_post')
    res['liveness'] = int(g('licence_canary') >= 1 and g('visits_canary') >= 1)
    res['exposure_ready'] = int(g('complete_towns_world') >= EXPOSE_TOWNS)   # WORLD evidence, never the completion row alone
    res['instrument_control'] = int(g('craft_post_control') >= 1 and g('gather_post_control') >= 1)
    for name, key in (('gather_ok', 'gather'), ('rung_ok', 'rung')):
        cells = [(g('%s_ok_pre_canary' % key), g('%s_pre_canary' % key)), (g('%s_ok_post_canary' % key), g('%s_post_canary' % key)),
                 (g('%s_ok_pre_control' % key), g('%s_pre_control' % key)), (g('%s_ok_post_control' % key), g('%s_post_control' % key))]
        d, z = did_z(cells)
        res['%s_did' % name] = d; res['%s_z' % name] = z
        res[name] = 1 if d is None or z is None or not (d < TOL and z < -2) else 0
    lit = lists.get('_litter') or {}
    can = [b - a for p, (a, b) in lit.items() if p in set(lists.get('_cans') or [])]
    con = [b - a for p, (a, b) in lit.items() if p not in set(lists.get('_cans') or [])]
    res['litter_did'] = (sum(can) / len(can) - sum(con) / len(con)) if can and con else None
    res['_lists'] = dict(lists)
    return res


def verdict_of(res, name):
    if res.get('%s_did' % name) is None or res.get('%s_z' % name) is None:
        return 'UNEXPOSED'
    return 'ok' if res[name] else 'DOWN'
